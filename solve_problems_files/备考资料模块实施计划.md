```python
"""MinerU 部署自检。用法：
    python backend/scripts/verify_mineru.py <sample.pdf>          # 完整OCR链路验收
    python backend/scripts/verify_mineru.py <sample.pdf> --text   # 仅测文字层
"""
import os, subprocess, sys, time
from pathlib import Path

MINERU_EXE = Path(__file__).resolve().parent.parent / "mineru_env" / "Scripts" / "mineru.exe"

def check_text_layer(pdf: str) -> int:
    import pdfplumber
    chars = pages = 0
    with pdfplumber.open(pdf) as doc:
        for p in doc.pages:
            chars += len((p.extract_text() or "").strip()); pages += 1
    avg = chars // max(pages, 1)
    print(f"[text-layer] pages={pages} total_chars={chars} avg/page={avg}")
    return avg

def run_mineru(pdf: str) -> float:
    if not MINERU_EXE.exists():
        sys.exit(f"[FAIL] mineru.exe not found: {MINERU_EXE}")
    out_dir = Path("./mineru_verify_out"); out_dir.mkdir(exist_ok=True)
    env = {**os.environ, "MINERU_MODEL_SOURCE": "modelscope"}
    t0 = time.time()
    r = subprocess.run([str(MINERU_EXE), "-p", pdf, "-o", str(out_dir), "-b", "pipeline"],
                       env=env, capture_output=True, text=True, timeout=900)
    cost = time.time() - t0
    mds = list(out_dir.rglob("*.md"))
    if r.returncode != 0 or not mds:
        print((r.stdout or "")[-1500:]); print((r.stderr or "")[-1500:])
        sys.exit(f"[FAIL] returncode={r.returncode} md_count={len(mds)}")
    content = mds[0].read_text(encoding="utf-8", errors="ignore")
    cjk = sum(1 for c in content[:5000] if "\u4e00" <= c <= "\u9fff")
    print(f"[OK] cost={cost:.1f}s md={mds[0]} cjk_in_head={cjk}")
    print(f"[BASELINE] pdf={pdf} seconds={cost:.1f} cjk_sample={cjk}")
    return cost

if __name__ == "__main__":
    pdf = sys.argv[1]
    if "--text" in sys.argv: check_text_layer(pdf)
    else: run_mineru(pdf)
```

```python
"""备考资料三表幂等迁移。用法：
    python backend/scripts/migrate_study_materials.py --dry-run   # 仅打印动作
    python backend/scripts/migrate_study_materials.py             # 执行（可安全重复运行）
"""
import argparse, sqlite3, sys
from pathlib import Path

DB = Path(__file__).resolve().parent.parent / "data" / "question_bank.db"

STUDY_MATERIAL_DDL = """
CREATE TABLE study_material (
    material_id INTEGER PRIMARY KEY AUTOINCREMENT,
    subject VARCHAR(20) NOT NULL,
    type VARCHAR(20) NOT NULL,
    category VARCHAR(50),
    title VARCHAR(200) NOT NULL,
    summary VARCHAR(500),
    content TEXT,
    card_front VARCHAR(200),
    card_back TEXT,
    cover_image VARCHAR(500),
    video_id INTEGER,
    duration INTEGER DEFAULT 0,
    read_time INTEGER DEFAULT 0,
    difficulty TINYINT DEFAULT 1,
    badge VARCHAR(20),
    sort_order INTEGER DEFAULT 0,
    status TINYINT DEFAULT 1,
    view_count INTEGER DEFAULT 0,
    study_count INTEGER DEFAULT 0,
    creator_id BIGINT REFERENCES sys_user(user_id),
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP
)"""

PROGRESS_DDL = """CREATE TABLE IF NOT EXISTS user_material_progress (
    progress_id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id BIGINT NOT NULL REFERENCES sys_user(user_id),
    material_id INTEGER NOT NULL REFERENCES study_material(material_id),
    status SMALLINT DEFAULT 0,
    progress_percent FLOAT DEFAULT 0,
    is_favorite BOOLEAN DEFAULT 0,
    completed_at DATETIME,
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, material_id)
)"""

IMPORT_DDL = """CREATE TABLE IF NOT EXISTS material_import_record (
    import_id INTEGER PRIMARY KEY AUTOINCREMENT,
    file_name VARCHAR(255) NOT NULL,
    file_type VARCHAR(10),
    file_size BIGINT,
    parse_mode VARCHAR(20),
    parse_status VARCHAR(20) DEFAULT 'pending',
    parse_result TEXT,
    error_message VARCHAR(1000),
    uploader_id BIGINT REFERENCES sys_user(user_id),
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP
)"""

IDX_DDL = [
    "CREATE INDEX IF NOT EXISTS idx_subject_type ON study_material(subject, type)",
    "CREATE INDEX IF NOT EXISTS idx_status_sort ON study_material(status, sort_order)",
    "CREATE INDEX IF NOT EXISTS idx_progress_user ON user_material_progress(user_id)",
    "CREATE INDEX IF NOT EXISTS idx_import_status ON material_import_record(parse_status)",
]

def table_exists(cur, name):
    return cur.execute(
        "SELECT 1 FROM sqlite_master WHERE type='table' AND name=?", (name,)
    ).fetchone() is not None

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--dry-run", action="store_true")
    args = ap.parse_args()
    if not DB.exists():
        sys.exit(f"[ABORT] db not found: {DB}")
    con = sqlite3.connect(DB)
    cur = con.cursor()
    actions = [("PRAGMA journal_mode=WAL",)]
    if table_exists(cur, "study_material"):
        n = cur.execute("SELECT COUNT(*) FROM study_material").fetchone()[0]
        if n > 0:
            sys.exit(f"[ABORT] study_material 已有 {n} 行！空表假设不成立，人工评估后再改脚本。")
        actions.append(("DROP TABLE study_material",))
    actions.append((STUDY_MATERIAL_DDL,))
    if not table_exists(cur, "user_material_progress"):
        actions.append((PROGRESS_DDL,))
    if not table_exists(cur, "material_import_record"):
        actions.append((IMPORT_DDL,))
    actions += [(d,) for d in IDX_DDL]

    print(f"[PLAN] {len(actions)} 个动作 @ {DB}")
    for a in actions:
        print("  -", a[0][:90].replace("\n", " "))
    if args.dry_run:
        print("[DRY-RUN] 未执行"); return
    for (sql,) in actions:
        con.execute(sql)
    con.commit()
    for t in ("study_material", "user_material_progress", "material_import_record"):
        assert table_exists(cur, t), f"missing {t}"
    mode = con.execute("PRAGMA journal_mode").fetchone()[0]
    print(f"[DONE] 三表就绪, journal_mode={mode}")

if __name__ == "__main__":
    main()
```

```python
"""备考资料 API。响应风格与 news.py 一致：裸 JSON。
学生组：GET "" 列表 / GET "/{id}" 详情 / GET "/progress-summary" / POST "/progress"
        POST "/{id}/favorite" / GET "/favorites"
管理组：T3/T6 追加于同文件下半部。
"""
import logging
from datetime import datetime
from typing import Optional

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy import func
from sqlalchemy.dialects.sqlite import insert as sqlite_insert
from sqlalchemy.orm import Session

from ...core.database import get_db
from ...core.security import get_current_user, get_optional_user
from ...models.models import StudyMaterial, UserMaterialProgress, SysUser

router = APIRouter(prefix="/api/materials", tags=["备考资料"])
logger = logging.getLogger(__name__)

SUBJECTS = {"xingce", "shenlun", "gongzhuan"}
TYPES = {"video_intro", "video_advanced", "article", "card"}


def _item_brief(m: StudyMaterial, prog=None) -> dict:
    d = {
        "id": m.material_id, "subject": m.subject, "type": m.type,
        "category": m.category, "title": m.title, "summary": m.summary,
        "badge": m.badge, "cover_image": m.cover_image,
        "duration": m.duration or 0, "read_time": m.read_time or 0,
        "difficulty": m.difficulty or 1, "view_count": m.view_count or 0,
        "card_front": m.card_front, "sort_order": m.sort_order or 0,
    }
    if prog is not None:
        d.update({
            "is_studied": prog.status == 2,
            "progress_percent": round(prog.progress_percent or 0, 1),
            "is_favorite": bool(prog.is_favorite),
        })
    return d


@router.get("", summary="资料列表(游客可访问)")
async def list_materials(
    subject: str = Query(..., description="xingce/shenlun/gongzhuan"),
    type: Optional[str] = Query(None),
    page: int = Query(1, ge=1), page_size: int = Query(20, ge=1, le=50),
    current_user: Optional[SysUser] = Depends(get_optional_user),
    db: Session = Depends(get_db),
):
    if subject not in SUBJECTS:
        raise HTTPException(400, "无效科目")
    if type and type not in TYPES:
        raise HTTPException(400, "无效类型")
    q = db.query(StudyMaterial).filter(
        StudyMaterial.subject == subject, StudyMaterial.status == 1)
    if type:
        q = q.filter(StudyMaterial.type == type)
    total = q.count()
    rows = (q.order_by(StudyMaterial.sort_order.asc(), StudyMaterial.material_id.desc())
             .offset((page - 1) * page_size).limit(page_size).all())
    progs = {}
    if current_user and rows:
        ids = [r.material_id for r in rows]
        for p in (db.query(UserMaterialProgress)
                  .filter(UserMaterialProgress.user_id == current_user.user_id,
                          UserMaterialProgress.material_id.in_(ids)).all()):
            progs[p.material_id] = p
    return {"total": total,
            "items": [_item_brief(m, progs.get(m.material_id)) for m in rows]}


@router.get("/progress-summary", summary="科目进度概览(登录)")
async def progress_summary(subject: str = Query(...),
                           current_user: SysUser = Depends(get_current_user),
                           db: Session = Depends(get_db)):
    if subject not in SUBJECTS:
        raise HTTPException(400, "无效科目")
    total = db.query(func.count(StudyMaterial.material_id)).filter(
        StudyMaterial.subject == subject, StudyMaterial.status == 1).scalar() or 0
    studied = (db.query(func.count(UserMaterialProgress.progress_id))
               .join(StudyMaterial,
                     StudyMaterial.material_id == UserMaterialProgress.material_id)
               .filter(UserMaterialProgress.user_id == current_user.user_id,
                       UserMaterialProgress.status == 2,
                       StudyMaterial.subject == subject,
                       StudyMaterial.status == 1).scalar() or 0)
    pct = round(studied * 100.0 / total, 1) if total else 0.0
    return {"studied": studied, "total": total, "percentage": pct}


@router.get("/favorites", summary="我的收藏(登录)")
async def my_favorites(current_user: SysUser = Depends(get_current_user),
                       db: Session = Depends(get_db)):
    rows = (db.query(StudyMaterial, UserMaterialProgress)
            .join(UserMaterialProgress,
                  UserMaterialProgress.material_id == StudyMaterial.material_id)
            .filter(UserMaterialProgress.user_id == current_user.user_id,
                    UserMaterialProgress.is_favorite.is_(True),
                    StudyMaterial.status == 1)
            .order_by(UserMaterialProgress.update_time.desc()).all())
    items = []
    for m, p in rows:
        item = _item_brief(m)
        item["favorite_time"] = p.update_time.isoformat() if p.update_time else None
        items.append(item)
    return {"total": len(rows), "items": items}


@router.post("/progress", summary="上报学习进度(登录, UPSERT幂等)")
async def report_progress(body: dict,
                          current_user: SysUser = Depends(get_current_user),
                          db: Session = Depends(get_db)):
    mid = body.get("material_id")
    if not mid:
        raise HTTPException(400, "缺少 material_id")
    try:
        pct = max(0.0, min(float(body.get("progress_percent", 0)), 100.0))
    except (TypeError, ValueError):
```

```python
"""备考资料多格式解析服务（同步轻量部分）。
docx -> 文章结构 | xlsx -> 行列数据(卡片列映射后置) | 文字型pdf -> 文章结构。
扫描件检测见 detect_scanned；MinerU 异步解析见 mineru_service(T5)。
"""
import os
import re
from pathlib import Path


def detect_scanned(pdf_path: str, threshold: int = 100) -> bool:
    """平均每页可提取字符 < threshold 判定为扫描件。"""
    import pdfplumber
    chars = pages = 0
    with pdfplumber.open(pdf_path) as doc:
        for page in doc.pages:
            chars += len((page.extract_text() or "").strip())
            pages += 1
    return (chars // max(pages, 1)) < threshold and pages > 0


def parse_docx(docx_path: str, image_dir: Path = None) -> dict:
    """Word -> {chapters:[{level,title}], html, word_count}
    内嵌图片写入 image_dir 并以 /uploads 相对路径引用（由调用方最终迁移）。"""
    from docx import Document
    from docx.table import Table
    from docx.text.paragraph import Paragraph

    doc = Document(docx_path)
    chapters, html_parts = [], []

    def _render_table(tbl) -> str:
        rows_html = []
        for r in tbl.rows:
            cells = "".join(f"<td>{c.text.strip()}</td>" for c in r.cells)
            rows_html.append(f"<tr>{cells}</tr>")
        return "<table><tbody>" + "".join(rows_html) + "</tbody></table>"

    def _extract_image(run, out_dir: Path) -> str:
        try:
            rid = run._element.xpath(".//a:blip/@r:embed")[0]
            part = doc.part.related_parts[rid]
            ext = os.path.splitext(part.partname)[1] or ".png"
            out_dir.mkdir(parents=True, exist_ok=True)
            name = f"{os.urandom(8).hex()}{ext}"
            (out_dir / name).write_bytes(part.blob)
            # 先返回临时可访问路径；confirm 阶段统一迁移到正式目录并改写
            return f'/uploads/materials/tmp/{name}'
        except Exception:
            return ""

    body = doc.element.body
    for child in body.iterchildren():
        if child.tag.endswith("}p"):
            para = Paragraph(child, doc)
            text = para.text.strip()
            style = (para.style.name or "").lower()
            if not text:
                html_parts.append("")
                continue
            if "heading 1" in style or "标题 1" in style:
                chapters.append({"level": 1, "title": text})
                html_parts.append(f"<h2>{text}</h2>")
            elif "heading 2" in style or "标题 2" in style:
                chapters.append({"level": 2, "title": text})
                html_parts.append(f"<h3>{text}</h3>")
            elif "heading 3" in style or "标题 3" in style:
                chapters.append({"level": 3, "title": text})
                html_parts.append(f"<h4>{text}</h4>")
            else:
                inner = ""
                for run in para.runs:
                    esc = (run.text or "").replace("&", "&amp;").replace("<", "&lt;")
                    if run.bold and esc.strip():
                        inner += f"<strong>{esc}</strong>"
                    else:
                        inner += esc
                    img = _extract_image(run, image_dir) if image_dir else ""
                    if img:
                        inner += f'<img src="{img}" alt=""/>'
                html_parts.append(f"<p>{inner or text}</p>")
        elif child.tag.endswith("}tbl"):
            html_parts.append(_render_table(Table(child, doc)))

    full_html = "".join(html_parts)
    word_count = len(re.sub(r"<[^>]+>", "", full_html))
    return {"chapters": chapters, "html": full_html, "word_count": word_count}


def parse_xlsx(xlsx_path: str, max_rows: int = 500) -> dict:
    """Excel -> {headers, rows(截断200行预览), total_rows}。列映射语义由 confirm 阶段决定。"""
    from openpyxl import load_workbook
    wb = load_workbook(xlsx_path, read_only=True, data_only=True)
    ws = wb.active
    rows_iter = ws.iter_rows(values_only=True)
    headers = []
    all_rows = []
    for i, row in enumerate(rows_iter):
        vals = [("" if v is None else str(v).strip()) for v in row]
        if i == 0:
            headers = vals
            continue
        if any(vals):
            all_rows.append(vals)
    wb.close()
    total = len(all_rows)
    return {"headers": headers, "rows": all_rows[:200], "total_rows": total,
            "over_limit": total > max_rows}


def build_cards(rows: list, col_front: int, col_back: int, col_category=None,
                limit: int = 500) -> list:
    """按列映射生成卡片数据。空行跳过；返回 [{card_front,card_back,category}]"""
    cards, seen = [], set()
    for r in rows:
        front = (r[col_front] or "").strip() if col_front < len(r) else ""
        back = (r[col_back] or "").strip() if col_back < len(r) else ""
        if not front or not back:
            continue
        key = (front, back)
        if key in seen:
            continue
        seen.add(key)
        cat = ""
        if col_category is not None and col_category < len(r):
            cat = (r[col_category] or "").strip()
        cards.append({"card_front": front[:200], "card_back": back, "category": cat})
    if len(cards) > limit:
        raise ValueError(f"有效卡片 {len(cards)} 张超过上限 {limit}，请拆分文件")
    return cards


def parse_pdf_text(pdf_path: str) -> dict:
    """文字型 PDF -> 文章结构。以字号聚类推断标题层级。"""
    import pdfplumber
    html_parts, chapters = [], []
    sizes = []
    pages_data = []
    with pdfplumber.open(pdf_path) as doc:
        for page in doc.pages:
            for ch in page.extract_words(extra_attrs=["size", "size_first_char"]):
                sizes.append(round(ch.get("size", 0), 1))
            pages_data.append(page.extract_words(extra_attrs=["size"]))
    body_size = sorted(sizes)[len(sizes) // 2] if sizes else 0
    h2_threshold = body_size * 1.3 if body_size else 0

    def flush_line(words):
        if not words:
            return
        text = "".join(w["text"] for w in words).strip()
        if not text:
            return
        size = max(w.get("size", 0) for w in words)
        if h2_threshold and size >= h2_threshold:
            chapters.append({"level": 1, "title": text})
            html_parts.append(f"<h2>{text}</h2>")
        else:
            html_parts.append(f"<p>{text}</p>")

    for words in pages_data:
        line, last_top = [], None
        for w in sorted(words, key=lambda x: (round(x["top"], 1), x["x0"])):
            if last_top is None or abs(w["top"] - last_top) <= 3:
                line.append(w)
            else:
                flush_line(line); line = [w]
            last_top = w["top"]
        flush_line(line)

    full = "".join(html_parts)
    return {"chapters": chapters, "html": full,
            "word_count": len(re.sub(r"<[^>]+>", "", full))}
```

```python
"""MinerU 子进程集成：扫描件 PDF -> Markdown -> 消毒 HTML。
进程隔离原则：MinerU(含torch)装在独立 venv backend/mineru_env，
主服务绝不 import 它，仅经 subprocess 调用 CLI。"""
import json
import re
import shutil
import subprocess
import time
from pathlib import Path

from .material_sanitizer import sanitize_html

BASE_DIR = Path(__file__).resolve().parent.parent.parent   # backend/
MINERU_EXE = BASE_DIR / "mineru_env" / "Scripts" / "mineru.exe"
TIMEOUT_SECONDS = 600


def mineru_available() -> bool:
    return MINERU_EXE.exists() and shutil.which(str(MINERU_EXE)) is not None


class MineruError(Exception):
    pass


class MineruTimeout(MineruError):
    pass


def run_mineru(input_path: str, work_dir: Path, timeout: int = TIMEOUT_SECONDS) -> dict:
    """阻塞执行（调用方须放入线程池/BackgroundTasks）。
    返回 {html, images:[{src_path,url}], seconds}；失败抛 MineruError/MineruTimeout。"""
    if not mineru_available():
        raise MineruError("OCR 组件未就绪")
    out_dir = work_dir / "mineru_out"
    out_dir.mkdir(parents=True, exist_ok=True)
    env = {**os.environ, "MINERU_MODEL_SOURCE": "modelscope"}
    t0 = time.time()
    try:
        r = subprocess.run(
            [str(MINERU_EXE), "-p", input_path, "-o", str(out_dir), "-b", "pipeline"],
            env=env, capture_output=True, text=True, timeout=timeout)
    except subprocess.TimeoutExpired:
        raise MineruTimeout(f"OCR 解析超时(>{timeout}s)")
    if r.returncode != 0:
        tail = (r.stderr or r.stdout or "")[-500:]
        raise MineruError(f"OCR 进程失败 rc={r.returncode}: {tail}")

    mds = list(out_dir.rglob("*.md"))
    if not mds:
        raise MineruError("OCR 未产出内容")
    md_text = mds[0].read_text(encoding="utf-8", errors="ignore")

    # Markdown -> HTML（标题/粗体/段落/图片，够用即可，复杂表格由 MinerU 输出为HTML片段时保留）
    html = _md_to_html(md_text)

    # 收集提取图片并改写引用为待迁移占位（confirm 时统一搬入正式目录）
    images = []
    img_root = mds[0].parent / "images"
    if img_root.exists():
        for f in img_root.iterdir():
            images.append({"src_path": str(f), "name": f.name})
    return {"html": sanitize_html(html),
            "images": images,
            "seconds": round(time.time() - t0, 1)}


def _md_to_html(md: str) -> str:
    parts = []
    for block in md.split("\n\n"):
        b = block.strip()
        if not b:
            continue
        b = re.sub(r"!\[([^\]]*)\]\(([^)]+)\)", r'<img src="\2" alt="\1"/>', b)
        if b.startswith("# "):
            parts.append(f"<h2>{b[2:].strip()}</h2>")
        elif b.startswith("## "):
            parts.append(f"<h3>{b[3:].strip()}</h3>")
        elif b.startswith("### "):
            parts.append(f"<h4>{b[4:].strip()}</h4>")
        else:
            b = re.sub(r"\*\*([^*]+)\*\*", r"<strong>\1</strong>", b)
            b = b.replace("\n", "<br/>")
            parts.append(f"<p>{b}</p>")
    return "".join(parts)
```

```vue
      return this.state.showFav ? arr.filter(i => i.is_favorite) :
             (this.state.folded[key] ? arr.slice(0, 4) : arr);
    },
    toggleFold(key) { this.state.folded[key] = !this.state.folded[key]; },
    async fetchAll() {
      for (const sec of this.sections()) {
        try {
          const r = await window.__VTJ_AUTH__.apiFetch(
            `/api/materials?subject=${SUBJECT}&type=${sec.key}&page_size=50`);
          const d = await r.json();
          this.state.data[sec.key] = d.items || [];
        } catch (e) {
          console.warn('[MatPage] load failed:', sec.key, e && e.message);
        } finally {
          this.state.loaded[sec.key] = true;
        }
      }
      this.state.pageEmpty =
        Object.values(this.state.data).every(a => !a.length);
    },
    async fetchSummary() {
      try {
        const auth = window.__VTJ_AUTH__ || {};
        this.state.loggedIn = !!(auth.isLoggedIn && auth.isLoggedIn());
        if (!this.state.loggedIn) return;
        const r = await window.__VTJ_AUTH__.apiFetch(
          `/api/materials/progress-summary?subject=${SUBJECT}`);
        if (r.ok) this.state.summary = await r.json();
      } catch (e) { console.warn('[MatPage] summary failed:', e && e.message); }
    },
    async ensureDetail(item) {
      // 弹窗需要 content/card_back/video_id —— 列表项不含时拉详情
      if (item.type === 'article' && item.content !== undefined) return item;
      const r = await window.__VTJ_AUTH__.apiFetch(`/api/materials/${item.id}`);
      const d = await r.json();
      Object.assign(item, d);
      return item;
    },
    async openVideo(it) { await this.ensureDetail(it); this.state.videoDlg = true; this.state.cur = it; },
    async openArticle(it) { await this.ensureDetail(it); this.state.articleDlg = true; this.state.cur = it; this.report(it, 5, false); },
    cardBack(it) { return it.card_back || ''; },
    flipCard(it) {
      this.state.flipped[it.id] = !this.state.flipped[it.id];
      if (this.state.flipped[it.id]) this.markDone(it);   // 首翻即完成（幂等由后端保证）
    },

      const v = e.target; if (!v.duration) return;
      const pct = Math.round(v.currentTime / v.duration * 100);
      this.report(this.state.cur, pct, pct >= 95);
    },
    report(item, pct, done) {
      if (!this.state.loggedIn) return;
      const now = Date.now(); const last = this.state._lastReport[item.id];
      const due = !last || now - last.ts >= 10000 || Math.abs(pct - last.pct) >= 20;
      if (!done && !due) return;
      this.state._lastReport[item.id] = { ts: now, pct };
      window.__VTJ_AUTH__.apiFetch('/api/materials/progress', {
        method: 'POST', headers: {'Content-Type':'application/json'},
        body: JSON.stringify({ material_id: item.id, progress_percent: pct, completed: done })
      }).then(r => r.json()).catch(() => {});
    },
    markDone(it) { this.report(it, 100, true); },
```

```vue
<script lang="ts">
// @ts-nocheck
import { defineComponent, reactive } from 'vue';
import { ElMessage } from 'element-plus';
import { useProvider } from '@vtj/renderer';

const TYPE_LABEL = {
  video_intro: '入门视频', video_advanced: '进阶视频',
  article: '文章', card: '知识卡片'
};

export default defineComponent({
  name: 'AdminMaterials',
  setup(props) {
    const provider = useProvider({ id: 'admin_materials', version: '20260825001' });
    const state = reactive({
      items: [], fSubject: '', fType: '',
      editDlg: false, saving: false, cardTab: 'single',
      form: {}, editor: null,
      xlsx: { headers: [], rows: [], total_rows: 0 },
      mapping: { front: 0, back: 1, category: null },
      uploadPct: 0, videoId: null,
      imp: { busy: false, id: null, file_type: '', parse_status: '', elapsed: 0,
             error: '', result: null, subject: 'xingce', type: 'article',
             category: '', mapping: { front: 0, back: 1, category: null }, confirming: false }
    });
    return { state, props, provider, TYPE_LABEL };
  },
  methods: {
    isVideo(t){ return t === 'video_intro' || t === 'video_advanced'; },
    api(url, opt) { return window.__VTJ_AUTH__.apiFetch(url, opt); },
    async loadList() {
      const q = new URLSearchParams();
      if (this.state.fSubject) q.set('subject', this.state.fSubject);
      if (this.state.fType) q.set('type', this.state.fType);
      const r = await this.api('/api/materials/admin/list?' + q.toString());
      const d = await r.json(); this.state.items = d.items || [];
    },
    openCreate() {
      this.state.form = { type: 'article', subject: 'xingce', title: '', summary: '',
        content: '', card_front: '', card_back: '', category: '', badge: '',
        duration: 0, difficulty: 1, cover_image: '' };
      this.state.videoId = null; this.state.uploadPct = 0;
      this.state.editDlg = true;
      if (this.state.cardTab === 'excel') this.state.cardTab = 'single';
    },
    async openEdit(row) {
      const r = await this.api(`/api/materials/${row.id}`); // 详情拿 content
      const d = await r.json();
      this.state.form = { ...row, ...d };
      this.state.editDlg = true;
    },
    async saveForm() {
      const f = this.state.form;
      if (!f.subject || !f.title) { ElMessage.warning('科目与标题必填'); return; }
      if (f.type === 'card' && !f.card_front && this.state.cardTab === 'single') {
        ElMessage.warning('卡片正面必填'); return;
      }
      if (this.isVideo(f.type) && !this.state.videoId && !f.video_id) {
        ElMessage.warning('请先上传视频'); return;
      }
      this.state.saving = true;
      try {
        const payload = { ...f };
        delete payload.is_studied; delete payload.progress_percent;
        delete payload.is_favorite; delete payload.content_preview;
        if (this.isVideo(f.type)) payload.video_id = f.video_id || this.state.videoId;
        if (f.type !== 'card') { delete payload.card_front; delete payload.card_back; }
        else { payload.content = f.card_back; }   // 卡片详述存 content 列（spec 3.1 注释约定）
        const url = f.id ? `/api/materials/admin/${f.id}` : '/api/materials/admin';
        const r = await this.api(url, { method: f.id ? 'PUT' : 'POST',
          headers: {'Content-Type':'application/json'},
          body: JSON.stringify(payload) });
        if (!r.ok) throw new Error((await r.json()).detail || '保存失败');
        ElMessage.success('已保存（默认下架态，请到列表开启上架）');
        this.destroyEditor(); this.state.editDlg = false; this.loadList();
      } catch (e) { ElMessage.error(e.message); }
      finally { this.state.saving = false; }
    },
    async setStatus(row, v) {
      const r = await this.api(`/api/materials/admin/${row.id}/status`,
        { method:'PUT', headers:{'Content-Type':'application/json'},
          body: JSON.stringify({ status: v ? 1 : 0 }) });
      if (r.ok) { row.status = v ? 1 : 0; ElMessage.success(v ? '已上架' : '已下架'); }
    },
    async del(row) {
      const r = await this.api(`/api/materials/admin/${row.id}`, { method: 'DELETE' });
      if (r.ok) { ElMessage.success('已删除'); this.loadList(); }
    },

    /* ---- wangeditor 动态挂载/销毁 ---- */
    async mountEditor() {
      if (!this.$refs || !this.$refs.editorEl) return;
      const E = (await import('wangeditor')).default;
      const ed = new E(this.$refs.editorEl);
      ed.config.height = 300;
      ed.config.customUploadImg = async (files, insert) => {
        for (const f of files) {
          const url = await this.uploadFileTo(f, '/api/materials/admin/upload/image');
          if (url) insert(url);
        }
      };
      ed.create();
      ed.txt.html(this.state.form.content || '');
      this.state.editor = ed;
    },
    destroyEditor() { if (this.state.editor) { this.state.editor.destroy(); this.state.editor = null; } },

    /* ---- 上传工具 ---- */
    async uploadFileTo(file, url) {
      const fd = new FormData(); fd.append('file', file);
      try {
        const r = await this.api(url, { method: 'POST', body: fd });
        const d = await r.json();
        if (!r.ok) throw new Error(d.detail || '上传失败');
        return d.url;
      } catch (e) { ElMessage.error(e.message); return ''; }
    },
    async uploadImage(e, field) {
      const url = await this.uploadFileTo(e.target.files[0], '/api/materials/admin/upload/image');
      if (url) this.state.form[field] = url;
    },

    /* ---- 视频分片上传（复用 videos.py 协议）----
       init -> chunk(file.slice 分片循环) -> complete；进度条按字节累计 */
    CHUNK_SIZE: 4 * 1024 * 1024,
    async pickVideo(e) {
      const file = e.target.files[0]; if (!file) return;
      this.state.uploadPct = 1;
      try {
        const ir = await this.api('/api/video/upload/init', {
          method: 'POST', headers: {'Content-Type':'application/json'},
          body: JSON.stringify({ filename: file.name, filesize: file.size }) });
        const init = await ir.json();
        const uploadId = init.data ? (init.data.upload_id ?? init.data.uploadId) : init.upload_id;
        let sent = 0; let index = 0;
        while (sent < file.size) {
          const blob = file.slice(sent, Math.min(sent + this.CHUNK_SIZE, file.size));
          const fd = new FormData();
          fd.append('upload_id', uploadId); fd.append('chunk_index', String(index));
          fd.append('chunk', blob);
          const cr = await this.api('/api/video/upload/chunk', { method: 'POST', body: fd });
          if (!cr.ok) throw new Error('分片上传失败');
          sent += blob.size; index += 1;
          this.state.uploadPct = Math.round(sent / file.size * 100);
        }
        const cr = await this.api('/api/video/upload/complete', {
          method: 'POST', headers: {'Content-Type':'application/json'},
          body: JSON.stringify({ upload_id: uploadId }) });
        const done = await cr.json();
        const vid = (done.data ? (done.data.video_id ?? done.data.id) : done.video_id)
                 ?? (done.data && done.data.id);
        this.state.videoId = vid;
        if (done.data && done.data.duration) this.state.form.duration = Math.round(done.data.duration);
        ElMessage.success('视频上传完成');
      } catch (err) { ElMessage.error(err.message); this.state.uploadPct = 0; }
    },

    /* ---- Excel 解析与批量 ---- */
    async parseExcel(e) {
      const file = e.target.files[0]; if (!file) return;
      // 复用导入通道：先 upload 拿到 rows（同步秒回），再走 confirm 的 column_mapping
      await this.startImport(file, 'xlsx');
    },
    mappingOk() {
      const m = this.state.mapping;
      return Number.isInteger(m.front) && Number.isInteger(m.back) && m.front !== m.back;
    },
    async confirmExcelCards() {
      // 走 import confirm（file_type=xlsx 时后端按 column_mapping 建卡）
      await this.doImportConfirm({
        subject: this.state.form.subject || 'xingce', type: 'card',
        column_mapping: this.state.mapping }, () => { this.state.editDlg = false; this.loadList(); });
    },

    /* ---- 智能导入向导 ---- */
    async startImport(file, forceType) {
      const imp = this.state.imp;
      Object.assign(imp, { busy: true, error: '', result: null, parse_status: 'uploading',
                           file_type: forceType || '', id: null, elapsed: 0 });
      const fd = new FormData(); fd.append('file', file);
      try {
        const r = await this.api('/api/materials/admin/import/upload', { method: 'POST', body: fd });
        const d = await r.json();
        if (!r.ok) throw new Error(d.detail || '上传失败');
        imp.id = d.import_id; imp.parse_status = d.parse_status;
        imp.file_type = d.file_type || imp.file_type;
        if (d.error_message) imp.error = d.error_message;
        if (imp.parse_status === 'processing') this.pollStatus();
        else if (imp.parse_status === 'success') await this.fetchPreview();
      } catch (err) { imp.error = err.message; }
      finally { imp.busy = false; }
    },
    async importPick(e) {
      const f = e.target.files[0]; if (!f) return;
      const ext = f.name.split('.').pop().toLowerCase();
      this.state.imp.file_type = ext;
      await this.startImport(f, ext);
    },
    pollStatus() {
      const imp = this.state.imp;
      const timer = setInterval(async () => {
        if (!imp.id || imp.parse_status !== 'processing') { clearInterval(timer); return; }
        const r = await this.api(`/api/materials/admin/import/${imp.id}/status`);
        const d = await r.json();
        imp.parse_status = d.parse_status; imp.elapsed = d.elapsed_seconds;
        if (d.error_message) imp.error = d.error_message;
        if (d.parse_status === 'success') await this.fetchPreview();
      }, 3000);
    },
    async fetchPreview() {
      const imp = this.state.imp;
      const r = await this.api(`/api/materials/admin/import/${imp.id}/preview`);
      if (!r.ok) return;
      const d = await r.json();
      imp.result = d.result;
    },
    canConfirmImport() {
      const imp = this.state.imp;
      return imp.parse_status === 'success' && !!imp.subject &&
             (imp.file_type === 'xlsx' ? this.impMappingOk() : true);
    },
    impMappingOk() {
      const m = this.state.imp.mapping;
      return Number.isInteger(m.front) && Number.isInteger(m.back);
    },
    async doImportConfirm(extraBody, after) {
      const imp = this.state.imp; imp.confirming = true;
      try {
        const body = { subject: imp.subject, type: imp.type,
                       category: imp.category, ...extraBody };
        if (imp.file_type === 'xlsx') body.column_mapping = extraBody.column_mapping || imp.mapping;
        const r = await this.api(`/api/materials/admin/import/${imp.id}/confirm`, {
          method: 'POST', headers: {'Content-Type':'application/json'},
          body: JSON.stringify(body) });
        const d = await r.json();
        if (!r.ok) throw new Error(d.detail || '入库失败');
        ElMessage.success(`成功创建 ${d.created} 条资料（下架态）`);
        Object.assign(imp, { parse_status: '', result: null, id: null });
        this.state.importDlg = false;
        if (after) after(); else this.loadList();
      } catch (e) { ElMessage.error(e.message); }
      finally { imp.confirming = false; }
    },
    async importConfirm() { await this.doImportConfirm({}); },
  },
  watch: {
    'state.editDlg'(v) {
      if (v && this.state.form.type === 'article')
        this.$nextTick(() => this.mountEditor());
      if (!v) this.destroyEditor();
    }
  },
  mounted() { this.loadList(); }
});
</script>
```
