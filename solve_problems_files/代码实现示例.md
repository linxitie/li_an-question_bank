@router.post("/detect-type", summary="统一题目类型识别")
async def detect_question_type(payload: Dict[str, Any]):
    """
    统一题目类型识别接口
    
    输入：
    {
        "question_content": "题目内容",
        "options": [{"key": "A", "value": "选项A"}, ...],
        "answer": "答案"
    }
    
    输出：
    {
        "question_type": "单选题",
        "confidence": 0.95,
        "reason": "基于答案格式、选项数量、题干关键词综合判断"
    }
    """
    try:
        content = payload.get('question_content', '')
        options = payload.get('options', [])
        answer = payload.get('answer', '')
        
        # 参数验证
        if not content:
            raise HTTPException(status_code=400, detail="题目内容不能为空")
        
        # 提取选项文本
        option_texts = []
        for opt in options:
            if isinstance(opt, dict):
                option_texts.append(opt.get('value', ''))
            else:
                option_texts.append(str(opt))
        
        # 调用后端识别逻辑
        from ...services.text_parser import _detect_question_type
        qtype, confidence = _detect_question_type(content, option_texts, answer)
        
        # 生成识别原因
        reason = _generate_type_reason(content, option_texts, answer, qtype, confidence)
        
        logger.info(f"[detect-type] 识别结果: {qtype} (置信度: {confidence:.2f})")
        
        return {
            "question_type": qtype,
            "confidence": round(confidence, 2),
            "reason": reason
        }
        
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"[detect-type] 识别失败: {e}", exc_info=True)
        raise HTTPException(status_code=500, detail=f"题目类型识别失败: {str(e)}")


def _generate_type_reason(content: str, options: List[str], answer: str, qtype: str, confidence: float) -> str:
    """生成题目类型识别原因"""
    reasons = []
    
    # 答案格式分析
    if answer:
        if re.match(r'^[A-D]$', answer):
            reasons.append("答案为单字母，判定为单选题")
        elif re.match(r'^[A-D]([，,][A-D])+$', answer):
            reasons.append("答案为多字母组合，判定为多选题")
        elif answer in ['正确', '错误', '对', '错', 'TRUE', 'FALSE', 'T', 'F']:
            reasons.append("答案为对/错，判定为判断题")
        elif len(answer) > 20:
            reasons.append("答案为长文本，判定为主观题")
    
    # 选项数量分析
    opt_count = len([o for o in options if o])
    if opt_count == 0:
        reasons.append("无选项，倾向主观题")
    elif opt_count == 2:
        reasons.append("2个选项，倾向判断题")
    elif opt_count == 4:
        reasons.append("4个选项，倾向单选题")
    elif opt_count > 4:
        reasons.append(f"{opt_count}个选项，倾向多选题")
    
    # 关键词分析
    if re.search(r'多选|多项选择|不定项', content):
        reasons.append("题干含多选关键词")
    if re.search(r'单选|选择题|下列.*正确', content):
        reasons.append("题干含单选关键词")
    if re.search(r'论述|分析|简答', content):
        reasons.append("题干含主观题关键词")
    
    if not reasons:
        reasons.append("综合各维度判断")
    
    return "；".join(reasons)

@router.get("/{paper_id}/preview", summary="试卷预览")
async def preview_paper(paper_id: int, db: Session = Depends(get_db)):
    """试卷预览接口，确保返回完整的题目内容"""
    
    # 1. 查询试卷基本信息
    paper = db.query(ExamPaper).filter(ExamPaper.paper_id == paper_id).first()
    if not paper:
        raise HTTPException(status_code=404, detail="试卷不存在")
    
    # 2. 查询关联的题目
    relations = db.query(PaperQuestionRelation).filter(
        PaperQuestionRelation.paper_id == paper_id
    ).order_by(PaperQuestionRelation.question_order).all()
    
    if not relations:
        return {
            "paper": {
                "paper_id": paper.paper_id,
                "paper_name": paper.paper_name,
                "question_count": 0,
                "total_score": paper.total_score
            },
            "questions": [],
            "message": "试卷无题目"
        }
    
    # 3. 查询题目详情
    question_ids = [r.question_id for r in relations]
    questions = db.query(QuestionBank).filter(
        QuestionBank.question_id.in_(question_ids)
    ).all()
    
    # 4. 构建题目映射
    question_map = {q.question_id: q for q in questions}
    
    # 5. 组装预览数据
    preview_questions = []
    for rel in relations:
        q = question_map.get(rel.question_id)
        if q:
            # 处理选项格式
            options = q.options
            if isinstance(options, str):
                try:
                    options = json.loads(options)
                except:
                    options = []
            
            preview_questions.append({
                "question_id": q.question_id,
                "question_content": q.question_content,
                "options": options,
                "answer": q.answer,
                "analysis": q.analysis,
                "question_type": q.question_type_detail,
                "score": rel.question_score,
                "order": rel.question_order
            })
    
    return {
        "paper": {
            "paper_id": paper.paper_id,
            "paper_name": paper.paper_name,
            "question_count": len(preview_questions),
            "total_score": paper.total_score,
            "create_time": paper.create_time.isoformat() if paper.create_time else None
        },
        "questions": preview_questions,
        "message": f"试卷包含 {len(preview_questions)} 道题目"
    }

// 修改 detectQuestionType 方法
async detectQuestionType(questionContent, options, answer = '') {
  var text = questionContent || '';
  var answerText = answer || '';
  
  // 1. 调用后端统一识别接口
  try {
    const res = await this.fetchWithRetry('/api/questions/import/detect-type', {
      method: 'POST',
      headers: this.getJsonHeaders(),
      body: JSON.stringify({
        question_content: questionContent,
        options: options,
        answer: answer
      })
    }, 5000, 1);
    const data = await res.json();
    if (res.ok && data.question_type) {
      console.log('[detectType] 后端识别结果:', data.question_type, '置信度:', data.confidence);
      return data.question_type;
    }
  } catch (e) {
    console.warn('[detectType] 后端识别失败，使用前端识别:', e);
  }
  
  // 2. 前端识别逻辑作为降级方案
  // 答案格式判断（权重最高）
  if (answerText) {
    if (/^[A-D]$/.test(answerText)) return '单选题';
    if (/^[A-D]([，,][A-D])+$/.test(answerText)) return '多选题';
    if (/^[A-D]{2,4}$/.test(answerText)) return '多选题';
    if (/^(正确|错误|对|错|TRUE|FALSE|T|F)$/.test(answerText)) return '判断题';
    if (answerText.length > 20) return '论述题';
  }
  
  // 选项数量判断
  if (!options || options.length < 2) {
    if (/写作|作文|请以|为题|写一篇|谈谈你的看法/.test(text) || text.length > 200) return '写作题';
    if (/论述|分析|简答|阐述|说明|概括/.test(text) || text.length > 80) return '论述题';
    if (/____|（）|\(\)|（\s*）/.test(text)) return '论述题';
    return '单选题';
  }
  
  // 选项内容分析
  var hasOptionMarkers = false;
  for (var i = 0; i < options.length; i++) {
    var opt = options[i];
    if (opt && (opt.key || opt.value)) {
      hasOptionMarkers = true;
      break;
    }
  }
  
  if (hasOptionMarkers) {
    if (options.length > 4) return '多选题';
    if (options.length === 4) return '单选题';
  }
  
  // 题干关键词判断
  if (/多选|多项选择|不定项/.test(text)) return '多选题';
  if (/单选|选择题|下列.*正确/.test(text)) return '单选题';
  if (/判断|对错|是否/.test(text)) return '判断题';
  
  // 默认单选题
  return '单选题';
}

// 新增验证题目类型的方法
async validateQuestionTypes(questions) {
  const validated = [];
  const total = questions.length;
  
  for (let i = 0; i < total; i++) {
    const q = questions[i];
    
    // 调用后端验证
    try {
      const typeRes = await this.fetchWithRetry('/api/questions/import/detect-type', {
        method: 'POST',
        headers: this.getJsonHeaders(),
        body: JSON.stringify({
          question_content: q.question_content,
          options: q.options,
          answer: q.answer || ''
        })
      }, 5000, 1);
      const typeData = await typeRes.json();
      if (typeRes.ok && typeData.question_type) {
        q.question_type_detail = typeData.question_type;
        q._type_confidence = typeData.confidence;
        q._type_reason = typeData.reason;
      }
    } catch (e) {
      console.warn('[validateTypes] 题目类型验证失败:', e);
    }
    
    validated.push(q);
    
    // 更新进度
    this.state.importProgress = Math.floor((i + 1) / total * 50);
  }
  
  return validated;
}

async importAllParsedQuestions() {
  const mode = this.state.importMode;
  var rate = Number(this.state.separateMatchRate || 0);
  
  // 匹配率分档
  if (mode === 'separate' && rate < 50) {
    ElMessage.error('匹配率仅 ' + rate + '%（低于50%）。请使用"手动匹配"界面建立对应关系后再导入。');
    this.showManualMatchDialog();
    return;
  }
  
  // 过滤掉已导入和标记为重复的题目
  const pending = this.state.parsedDocsQuestions.filter(function(q) {
    return !q._imported && !q._is_duplicate;
  });
  
  if (!pending.length) {
    var already = this.state.parsedDocsQuestions.filter(function(q) { return q._imported; }).length;
    var dups = this.state.parsedDocsQuestions.filter(function(q) { return q._is_duplicate; }).length;
    if (dups > 0) {
      ElMessage.info('所有新题目已处理完毕（已导入 ' + already + ' 题，跳过 ' + dups + ' 题重复）');
    } else {
      ElMessage.info('所有题目已导入');
    }
    return;
  }
  
  if (mode === 'combined') {
    const noAnswer = pending.filter(function(q) { return !(q.answer || '').trim() || q.answer === '待补充'; });
    if (noAnswer.length) {
      ElMessage.warning('合并模式下有 ' + noAnswer.length + ' 道题目缺少答案，请填写后导入或切换到分离导入模式');
      return;
    }
  }
  
  this.state.loading = true;
  this.state.importProgress = 0;
  
  try {
    // ===== 新增：验证题目类型 =====
    ElMessage.info('正在验证题目类型...');
    const validated = await this.validateQuestionTypes(pending);
    
    // ===== 单请求批量入库 =====
    ElMessage.info('正在导入题目...');
    this.state.importProgress = 50;
    
    const res = await this.fetchWithRetry('/api/questions/import/batch-parsed', {
      method: 'POST',
      headers: this.getJsonHeaders(),
      body: JSON.stringify({
        paper_name: this.state.batchPaperName || (mode === 'separate' ? '分离导入试卷' : 'Word批量导入'),
        subject_category: (validated[0] && validated[0].subject_category) || '公专',
        import_mode: mode,
        match_rate: mode === 'separate' ? rate : 0,
        questions: validated.map(function(q) {
          return {
            question_content: q.question_content,
            options: q.options,
            answer: q.answer || '',
            answer_analysis: q.analysis || '',
            question_type_detail: q.question_type_detail || '单选题',
            subject_category: q.subject_category,
            score: q.score || 1,
            difficulty: q.difficulty || 3
          };
        })
      })
    }, 600000);
    
    const data = await res.json();
    if (!res.ok) {
      ElMessage.error(data.detail || ('批量导入失败 (HTTP ' + res.status + ')'));
    } else {
      this.state.duplicateCount = data.duplicate_count || 0;
      validated.forEach(function(q) { q._imported = true; });
      
      this.state.importSummary = {
        total: data.total || 0,
        success: data.success_count || 0,
        dup: data.duplicate_count || 0,
        failed: (data.failed_items || []).length,
        needConfirm: data.need_confirm_count || 0,
        matchRate: data.match_rate || rate,
        paperId: data.paper_id,
        paperName: data.paper_name || this.state.batchPaperName || '',
        mode: mode
      };
      
      if ((data.failed_items || []).length) {
        console.warn('[batch-parsed] 失败明细:', data.failed_items);
      }
      
      this.state.importProgress = 100;
      ElMessage.success(data.message || ('导入完成：成功 ' + data.success_count + ' 题'));
      
      if (data.need_confirm_count > 0) {
        ElMessage.warning(data.need_confirm_count + ' 题缺答案/低置信度，建议在试卷编辑中人工确认后再发布');
      }
      
      // 新增：验证试卷预览
      if (data.paper_id) {
        await this.verifyPaperPreview(data.paper_id);
      }
    }
  } catch (e) {
    ElMessage.error(this.handleFetchError(e, '批量导入'));
  }
  
  this.loadRecords();
  this.notifyBankRefresh();
  this.state.loading = false;
  this.state.importProgress = 0;
}
