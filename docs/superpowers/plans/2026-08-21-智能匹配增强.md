# backend/app/services/text_parser.py (添加到文件开头)
import re

class TextPreprocessor:
    def clean_text(self, text: str) -> str:
        """
        文本预处理流程：
        1. 统一编码：UTF-8
        2. 统一字符：全角→半角，中文标点→英文标点
        3. 清理空白：多余空格、换行、制表符
        4. 去除干扰：页眉、页脚、水印、页码
        5. 统一题号格式：所有题号统一为"数字."格式
        """
        if not text:
            return ""
        
        # 1. 清理空白
        text = re.sub(r'\s+', ' ', text)
        text = text.strip()
        
        # 2. 去除页码
        text = re.sub(r'第\s*\d+\s*页', '', text)
        text = re.sub(r'\d+\s*页', '', text)
        
        # 3. 去除页眉页脚（常见格式）
        text = re.sub(r'^(.*?)(?:页眉|页脚|水印).*?$', '', text, flags=re.MULTILINE)
        
        return text
    
    def normalize_question_numbers(self, text: str) -> str:
        """
        统一题号格式：
        - "一、" → "1."
        - "（1）" → "1."
        - "第1题" → "1."
        - "Q1:" → "1."
        """
        if not text:
            return ""
        
        # 中文数字映射
        cn_num_map = {
            '一': '1', '二': '2', '三': '3', '四': '4', '五': '5',
            '六': '6', '七': '7', '八': '8', '九': '9', '十': '10',
        }
        
        # 替换中文数字题号
        for cn, num in cn_num_map.items():
            text = re.sub(f'{cn}、', f'{num}.', text)
        
        # 替换其他格式
        text = re.sub(r'（(\d+)）', r'\1.', text)
        text = re.sub(r'第(\d+)题', r'\1.', text)
        text = re.sub(r'Q(\d+):', r'\1.', text)
        
        return text
```

# backend/app/services/text_parser.py (在TextPreprocessor类之后添加)
class QuestionNumberRecognizer:
    # 支持的题号格式
    PATTERNS = [
        r'(\d+)\.',           # 1.
        r'(\d+)、',           # 1、
        r'(\d+)．',           # 1．
        r'(\d+)',             # 1）
        r'（(\d+)）',         # （1）
        r'\[(\d+)\]',         # [1]
        r'第(\d+)题',         # 第1题
        r'Q(\d+):',           # Q1:
        r'问题(\d+)',         # 问题1
        r'题(\d+)',           # 题1
    ]
    
    # 中文数字映射
    CN_NUM_MAP = {
        '一': 1, '二': 2, '三': 3, '四': 4, '五': 5,
        '六': 6, '七': 7, '八': 8, '九': 9, '十': 10,
        '十一': 11, '十二': 12, '十三': 13, '十四': 14, '十五': 15,
        '十六': 16, '十七': 17, '十八': 18, '十九': 19, '二十': 20,
    }
    
    def recognize_with_confidence(self, text: str) -> list:
        """
        带置信度的题号识别
        返回：[{
            'number': int,           # 题号
            'position': int,         # 位置
            'format': str,           # 格式类型
            'confidence': float,     # 置信度 0-1
            'context': str,          # 上下文
        }]
        """
        if not text:
            return []
        
        results = []
        lines = text.split('\n')
        
        for line_num, line in enumerate(lines):
            for pattern in self.PATTERNS:
                match = re.search(pattern, line)
                if match:
                    number = int(match.group(1))
                    confidence = 0.8  # 基础置信度
                    
                    # 如果是连续题号，提高置信度
                    if results and number == results[-1]['number'] + 1:
                        confidence = 0.95
                    
                    results.append({
                        'number': number,
                        'position': line_num,
                        'format': pattern,
                        'confidence': confidence,
                        'context': line[:50],
                    })
                    break
        
        return results
    
    def resolve_conflicts(self, candidates: list) -> list:
        """
        解决题号识别冲突
        使用上下文信息和连续性判断选择最佳匹配
        """
        if not candidates:
            return []
        
        # 按位置排序
        sorted_candidates = sorted(candidates, key=lambda x: x['position'])
        
        # 选择连续性最好的序列
        best_sequence = []
        current_sequence = []
        
        for candidate in sorted_candidates:
            if not current_sequence:
                current_sequence.append(candidate)
            elif candidate['number'] == current_sequence[-1]['number'] + 1:
                current_sequence.append(candidate)
            else:
                if len(current_sequence) > len(best_sequence):
                    best_sequence = current_sequence
                current_sequence = [candidate]
        
        if len(current_sequence) > len(best_sequence):
            best_sequence = current_sequence
        
        return best_sequence
```

# backend/app/services/text_parser.py (在QuestionNumberRecognizer类之后添加)
class AnswerExtractor:
    # 支持的答案格式
    ANSWER_PATTERNS = [
        r'答案[：:]\s*([A-Da-d]+)',           # 答案：A
        r'参考答案[：:]\s*([A-Da-d]+)',       # 参考答案：AB
        r'正确答案[：:]\s*([A-Da-d]+)',       # 正确答案：ABC
        r'标准答案[：:]\s*([A-Da-d]+)',       # 标准答案：ABCD
        r'Answer[：:]\s*([A-Da-d]+)',         # Answer: A
        r'故本题答案为([A-Da-d]+)',           # 故本题答案为D
        r'本题选([A-Da-d]+)',                 # 本题选AB
        r'应选([A-Da-d]+)',                   # 应选C
    ]
    
    def extract_with_context(self, text: str, question_block: dict) -> dict:
        """
        带上下文的答案提取
        输入：题目文本和题目块信息
        输出：提取的答案和置信度
        """
        if not text:
            return {'answer': '', 'confidence': 0.0, 'format': '', 'raw_text': ''}
        
        question_type = question_block.get('question_type', '单选题')
        
        # 尝试所有答案格式
        for pattern in self.ANSWER_PATTERNS:
            match = re.search(pattern, text, re.IGNORECASE)
            if match:
                answer = match.group(1).upper()
                
                # 验证答案
                if self.validate_answer(answer, question_type):
                    return {
                        'answer': answer,
                        'confidence': 0.9,
                        'format': pattern,
                        'raw_text': match.group(0),
                    }
        
        # 如果没有找到标准格式，尝试提取单个字母
        single_letter_match = re.search(r'\b([A-Da-d])\b', text)
        if single_letter_match:
            answer = single_letter_match.group(1).upper()
            if self.validate_answer(answer, question_type):
                return {
                    'answer': answer,
                    'confidence': 0.7,
                    'format': 'single_letter',
                    'raw_text': single_letter_match.group(0),
                }
        
        return {'answer': '', 'confidence': 0.0, 'format': '', 'raw_text': ''}
    
    def validate_answer(self, answer: str, question_type: str) -> bool:
        """
        验证答案是否合理
        根据题目类型验证答案格式
        """
        if question_type in ('单选题', '多选题'):
            # 验证选择题答案格式
            return self._validate_choice_answer(answer, question_type)
        elif question_type in ('论述题', '写作题'):
            # 验证论述题/写作题答案：至少50个字符
            return len(answer.strip()) >= 50
        else:
            # 其他题型：只要不为空即可
            return bool(answer.strip())
    
    def _validate_choice_answer(self, answer: str, question_type: str) -> bool:
        """
        验证选择题答案格式
        - 单选题：单个字母A-D
        - 多选题：多个字母A-D，按字母顺序排列
        """
        # 去除空格和标点
        clean_answer = re.sub(r'[\s、,，]', '', answer.upper())
        
        if question_type == '单选题':
            # 单选题：单个字母A-D
            return bool(re.match(r'^[A-D]$', clean_answer))
        elif question_type == '多选题':
            # 多选题：多个字母A-D，按字母顺序排列
            if not re.match(r'^[A-D]+$', clean_answer):
                return False
            # 检查是否按字母顺序排列
            return clean_answer == ''.join(sorted(clean_answer))
        return False
```

# backend/app/services/similarity.py
import re
from difflib import SequenceMatcher
from collections import Counter
import math

class SimilarityCalculator:
    def calculate_edit_distance(self, s1: str, s2: str) -> float:
        """
        计算编辑距离相似度
        基于Levenshtein距离，计算字符级相似度
        """
        if not s1 and not s2:
            return 1.0
        if not s1 or not s2:
            return 0.0
        
        # 使用SequenceMatcher计算相似度
        matcher = SequenceMatcher(None, s1, s2)
        return matcher.ratio()
    
    def calculate_cosine_similarity(self, s1: str, s2: str) -> float:
        """
        计算余弦相似度
        基于TF-IDF向量化，计算语义相似度
        """
        if not s1 and not s2:
            return 1.0
        if not s1 or not s2:
            return 0.0
        
        # 简单分词（按字符）
        words1 = list(s1)
        words2 = list(s2)
        
        # 计算词频
        counter1 = Counter(words1)
        counter2 = Counter(words2)
        
        # 计算余弦相似度
        intersection = set(counter1.keys()) & set(counter2.keys())
        numerator = sum([counter1[x] * counter2[x] for x in intersection])
        
        sum1 = sum([counter1[x] ** 2 for x in counter1.keys()])
        sum2 = sum([counter2[x] ** 2 for x in counter2.keys()])
        denominator = math.sqrt(sum1) * math.sqrt(sum2)
        
        if not denominator:
            return 0.0
        
        return float(numerator) / denominator
    
    def calculate_jaccard_similarity(self, s1: str, s2: str) -> float:
        """
        计算Jaccard相似度
        基于词汇重叠度，计算词汇相似度
        """
        if not s1 and not s2:
            return 1.0
        if not s1 or not s2:
            return 0.0
        
        # 简单分词（按字符）
        set1 = set(s1)
        set2 = set(s2)
        
        intersection = len(set1 & set2)
        union = len(set1 | set2)
        
        if union == 0:
            return 0.0
        
        return intersection / union
    
    def calculate_sequence_similarity(self, s1: str, s2: str) -> float:
        """
        计算序列相似度
        基于最长公共子序列，计算结构相似度
        """
        if not s1 and not s2:
            return 1.0
        if not s1 or not s2:
            return 0.0
        
        # 使用SequenceMatcher计算相似度
        matcher = SequenceMatcher(None, s1, s2)
        return matcher.ratio()
    
    def calculate_combined_similarity(self, s1: str, s2: str) -> dict:
        """
        综合多种相似度指标
        返回：{
            'edit_distance': float,      # 编辑距离相似度
            'cosine_similarity': float,  # 余弦相似度
            'jaccard_similarity': float, # Jaccard相似度
            'sequence_similarity': float,# 序列相似度
            'combined_score': float,     # 综合得分
        }
        """
        edit_distance = self.calculate_edit_distance(s1, s2)
        cosine_similarity = self.calculate_cosine_similarity(s1, s2)
        jaccard_similarity = self.calculate_jaccard_similarity(s1, s2)
        sequence_similarity = self.calculate_sequence_similarity(s1, s2)
        
        # 综合得分（加权平均）
        combined_score = (
            edit_distance * 0.3 +
            cosine_similarity * 0.3 +
            jaccard_similarity * 0.2 +
            sequence_similarity * 0.2
        )
        
        return {
            'edit_distance': edit_distance,
            'cosine_similarity': cosine_similarity,
            'jaccard_similarity': jaccard_similarity,
            'sequence_similarity': sequence_similarity,
            'combined_score': combined_score,
        }
```

class StructureMatcher:
    def extract_structure_features(self, question_block: dict) -> dict:
        """
        提取题目结构特征（增强版）
        返回：{
            'question_length': int,      # 题目长度
            'option_count': int,         # 选项数量
            'option_type': str,          # 选项类型：'numeric'/'text'/'mixed'
            'option_format': str,        # 选项格式：'numbered'/'plain'
            'has_image': bool,           # 是否包含图片

            'paragraph_count': int,      # 段落数量
            'sentence_count': int,       # 句子数量
            'word_count': int,           # 词数
            'avg_sentence_length': float,# 平均句子长度
            'punctuation_type': str,     # 标点类型：'chinese'/'english'/'mixed'
            'question_type': str,        # 题目类型：'single'/'multi'/'judge'/'essay'
        }
        """
        content = question_block.get('content', '')
        options = question_block.get('options', [])
        question_type = question_block.get('question_type', '单选题')
        
        # 计算题目长度
        question_length = len(content)
        
        # 计算选项数量
        option_count = len(options)
        
        # 判断选项类型
        option_type = self._detect_option_type(options)
        
        # 判断选项格式
        option_format = self._detect_option_format(options)
        
        # 检测是否包含图片
        has_image = bool(re.search(r'<img|IMAGE|\.(jpg|jpeg|png|gif)', content, re.IGNORECASE))
        
        # 检测是否包含公式
        has_formula = bool(re.search(r'\$|\\frac|\\sum|\\int|数学公式', content, re.IGNORECASE))
        
        # 计算段落数量
        paragraph_count = len([p for p in content.split('\n') if p.strip()])
        
        # 计算句子数量
        sentences = re.split(r'[。！？.!?]', content)
        sentence_count = len([s for s in sentences if s.strip()])
        
        # 计算词数
        word_count = len(content)
        
        # 计算平均句子长度
        avg_sentence_length = word_count / sentence_count if sentence_count > 0 else 0.0
        
        # 判断标点类型
        punctuation_type = self._detect_punctuation_type(content)
        
        # 映射题目类型
        question_type_map = {
            '单选题': 'single',
            '多选题': 'multi',
            '判断题': 'judge',
            '论述题': 'essay',
            '写作题': 'essay',
        }
        mapped_question_type = question_type_map.get(question_type, 'single')
        
        return {
            'question_length': question_length,
            'option_count': option_count,
            'option_type': option_type,
            'option_format': option_format,
            'has_image': has_image,
            'has_formula': has_formula,
            'paragraph_count': paragraph_count,
            'sentence_count': sentence_count,
            'word_count': word_count,
            'avg_sentence_length': avg_sentence_length,
            'punctuation_type': punctuation_type,
            'question_type': mapped_question_type,
        }
    
    def _detect_option_type(self, options: list) -> str:
        """检测选项类型"""
        if not options:
            return 'none'
        
        numeric_count = 0
        text_count = 0
        
        for option in options:
            value = option.get('value', '')
            if re.match(r'^\d+\.?\d*$', value):
                numeric_count += 1
            else:
                text_count += 1
        
        if numeric_count > text_count:
            return 'numeric'
        elif text_count > numeric_count:
            return 'text'
        else:
            return 'mixed'
    
    def _detect_option_format(self, options: list) -> str:
        """检测选项格式"""
        if not options:
            return 'none'
        
        numbered_count = 0
        plain_count = 0
        
        for option in options:
            key = option.get('key', '')
            if re.match(r'^[A-D]$', key):
                numbered_count += 1
            else:
                plain_count += 1
        
        if numbered_count > plain_count:
            return 'numbered'
        else:
            return 'plain'
    
    def _detect_punctuation_type(self, text: str) -> str:
        """检测标点类型"""
        chinese_punct = len(re.findall(r'[\u3000-\u303F\uFF00-\uFFEF]', text))
        english_punct = len(re.findall(r'[!-/:-@[-`{-~]', text))
        
        if chinese_punct > english_punct:
            return 'chinese'
        elif english_punct > chinese_punct:
            return 'english'
        else:
            return 'mixed'
    
    def calculate_structure_similarity(self, features1: dict, features2: dict) -> float:
        """
        计算结构特征相似度（增强版）
        考虑题目类型、选项内容、题目格式、题目来源等因素
        """
        if not features1 or not features2:
            return 0.0
        
        # 计算各个特征的相似度
        similarities = []
        
        # 题目长度相似度
        length_sim = 1.0 - abs(features1['question_length'] - features2['question_length']) / max(features1['question_length'], features2['question_length'], 1)
        similarities.append(length_sim * 0.2)
        
        # 选项数量相似度
        option_count_sim = 1.0 - abs(features1['option_count'] - features2['option_count']) / max(features1['option_count'], features2['option_count'], 1)
        similarities.append(option_count_sim * 0.3)
        
        # 选项类型相似度
        option_type_sim = 1.0 if features1['option_type'] == features2['option_type'] else 0.5
        similarities.append(option_type_sim * 0.2)
        
        # 题目类型相似度
        question_type_sim = 1.0 if features1['question_type'] == features2['question_type'] else 0.3
        similarities.append(question_type_sim * 0.3)
        
        # 计算综合相似度
        total_weight = sum([0.2, 0.3, 0.2, 0.3])
        weighted_sum = sum(similarities)
        
        return weighted_sum / total_weight if total_weight > 0 else 0.0
    
    def match_by_structure(self, question_blocks: list, answer_blocks: list) -> dict:
        """
        基于结构特征进行匹配
        返回：{
            'matches': list,      # 匹配结果
            'confidence': float,  # 置信度
        }
        """
        matches = []
        used_answers = set()
        
        for q_block in question_blocks:
            q_features = self.extract_structure_features(q_block)
            best_match = None
            best_similarity = 0.0
            
            for i, a_block in enumerate(answer_blocks):
                if i in used_answers:
                    continue
                
                a_features = self.extract_structure_features(a_block)
                similarity = self.calculate_structure_similarity(q_features, a_features)
                
                if similarity > best_similarity:
                    best_similarity = similarity
                    best_match = i
            
            if best_match is not None and best_similarity > 0.7:
                matches.append({
                    'question': q_block,
                    'answer': answer_blocks[best_match],
                    'similarity': best_similarity,
                })
                used_answers.add(best_match)
        
        confidence = len(matches) / len(question_blocks) if question_blocks else 0.0
        
        return {
            'matches': matches,
            'confidence': confidence,
        }
