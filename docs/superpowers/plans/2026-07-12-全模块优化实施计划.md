```python
import os
import hashlib
from ..models.models import QuestionImage

def _extract_and_save_images(content_html, question_id, upload_dir='uploads/questions'):
    """
    从HTML内容中提取base64或url图片，保存到磁盘，写入question_image表
    返回图片路径列表
    """
    import re
    import base64
    from io import BytesIO
    from PIL import Image

    images = []
    img_dir = os.path.join(upload_dir, str(question_id))
    os.makedirs(img_dir, exist_ok=True)

    # 提取 <img> 标签
    img_tags = re.findall(r'<img[^>]+src=["\']([^"\']+)["\'][^>]*>', content_html)

    for idx, src in enumerate(img_tags):
        try:
            if src.startswith('data:image'):
                # Base64图片
                header, data = src.split(',', 1)
                img_data = base64.b64decode(data)
                ext = header.split(';')[0].split('/')[-1] or 'png'
                filename = f'img_{idx}.{ext}'
                filepath = os.path.join(img_dir, filename)
                with open(filepath, 'wb') as f:
                    f.write(img_data)
            elif src.startswith('http'):
                # URL图片 — 下载
                import requests
                resp = requests.get(src, timeout=10)
                if resp.status_code == 200:
                    ext = src.rsplit('.', 1)[-1].split('?')[0] or 'jpg'
                    filename = f'img_{idx}.{ext}'
                    filepath = os.path.join(img_dir, filename)
                    with open(filepath, 'wb') as f:
                        f.write(resp.content)
                else:
                    continue
            else:
                continue

            # 存储路径（相对路径用于前端访问）
            rel_path = f'/static/questions/{question_id}/{filename}'
            images.append(rel_path)

        except Exception as e:
            logger.warning(f"[images] 图片{idx}保存失败: {e}")
            continue

    return images
```

from app.models.models import QuestionBank, QuestionDetailCategory

import logging

def migrate(dry_run=False):
    db = SessionLocal()
    try:
        # 查询所有缺少QuestionDetailCategory的题目
        from sqlalchemy import and_
        questions = db.query(QuestionBank).outerjoin(
            QuestionDetailCategory,
            QuestionBank.question_id == QuestionDetailCategory.question_id
        ).filter(
            QuestionDetailCategory.category_id == None
        ).all()

        logger.info(f"找到 {len(questions)} 条待回填题目")

        count = 0
        for q in questions:
            # 推断题型
            if q.answer and len(q.answer.strip()) > 1:
                qtype = '多选题'
            else:
                # 从题目文本检测
                options = q.options or []
                if isinstance(options, str):
                    import json
                    try:
                        options = json.loads(options)
                    except:
                        options = []

                    question_id=q.question_id,
                    question_type_detail=qtype,
                    xingce_subcategory='无',
                    shenlun_subcategory='无',

        logger.info(f"{'[DRY-RUN] 将' if dry_run else '已'}回填 {count} 条题目")

- [ ] **Step 1: 创建难度评估引擎**

```python
"""
渐进式IRT难度评估引擎
冷启动: 正确率法 (<30次答题)
预热期: Rasch 1PL (30-200次)
成熟期: 2PL (>200次)
"""
import logging
import math
from typing import Dict, List, Tuple, Optional
from datetime import datetime

logger = logging.getLogger(__name__)


def estimate_difficulty_cold_start(
    correct_count: int,
    total_count: int
) -> int:
    """
    冷启动阶段：基于正确率估算难度 (1-5级)
    difficulty = round(5 * (1 - accuracy))
    """
    if total_count == 0:
        return 3  # 默认中等
    accuracy = correct_count / total_count
    raw = 5 * (1 - accuracy)
    return max(1, min(5, round(raw)))


def estimate_difficulty_rasch(
    correct_count: int,
    total_count: int,
    student_abilities: List[float] = None
) -> float:
    """
    Rasch 1PL模型：仅估计难度参数 b
    使用MLE (Maximum Likelihood Estimation)
    P(correct|ability) = 1 / (1 + exp(-(ability - b)))
    """
    if total_count < 30:
        return 3.0

    # 简化版Rasch: b = -ln(accuracy / (1 - accuracy)) 归一化到1-5范围
    accuracy = max(0.01, min(0.99, correct_count / total_count))
    logit = -math.log(accuracy / (1 - accuracy))

    # 映射logit到1-5范围
    # logit ∈ [-4, 4] → difficulty ∈ [1, 5]
    normalized = (logit + 4) / 8 * 4 + 1
    return max(1.0, min(5.0, normalized))


def estimate_difficulty_2pl(
    correct_count: int,
    total_count: int,
    student_correct_counts: List[int],
    student_total_counts: List[int]
) -> Tuple[float, float]:
    """
    2PL模型：估计难度 b 和区分度 a
    使用scipy.optimize进行MLE估计

    P(correct|ability, a, b) = 1 / (1 + exp(-a * (ability - b)))
    """
    try:
        from scipy.optimize import minimize
        import numpy as np
    except ImportError:
        logger.warning("scipy未安装，降级为Rasch")
        b = estimate_difficulty_rasch(correct_count, total_count)
        return b, 1.0

    if total_count < 200 or len(student_correct_counts) < 10:
        b = estimate_difficulty_rasch(correct_count, total_count)
        return b, 1.0

    # 估计每个学生的能力（简化：正确率作为能力近似）
    abilities = []
    for cc, tc in zip(student_correct_counts, student_total_counts):
        if tc > 0:
            acc = max(0.01, min(0.99, cc / tc))
            abilities.append(-math.log(acc / (1 - acc)))
        else:
            abilities.append(0.0)
    abilities = np.array(abilities)

    # MLE估计 a 和 b
    responses = np.array([1] * correct_count + [0] * (total_count - correct_count))

    def neg_log_likelihood(params):
        a, b = params
        # 对每个学生计算P(correct)并求和log-likelihood
        probs = 1 / (1 + np.exp(-a * (abilities - b)))
        # 为避免log(0)，钳制概率
        probs = np.clip(probs, 1e-10, 1 - 1e-10)
        # 使用总体的正确/错误比例加权
        ll = correct_count * np.log(probs.mean()) + \
             (total_count - correct_count) * np.log(1 - probs.mean())
        return -ll

    result = minimize(neg_log_likelihood, x0=[1.0, 0.0],
                      bounds=[(0.3, 3.0), (-4.0, 4.0)])

    a, b = result.x
    # 映射b到1-5范围
    difficulty = max(1.0, min(5.0, (b + 4) / 8 * 4 + 1))
    return round(difficulty, 2), round(a, 2)


def get_difficulty_level(question_id: int, db_session) -> int:
    """
    获取题目难度等级（主入口）
    自动选择冷启动/Rasch/2PL阶段
    """
    from ..models.models import PracticeRecord

    # 统计答题记录
    records = db_session.query(PracticeRecord).filter(
        PracticeRecord.question_id == question_id
    ).all()

    total = len(records)
    correct = sum(1 for r in records if r.is_correct)

    if total < 30:
        return estimate_difficulty_cold_start(correct, total)
    elif total < 200:
        return round(estimate_difficulty_rasch(correct, total))
    else:
        b, a = estimate_difficulty_2pl(correct, total, [], [])
        return round(b)


def batch_update_difficulties(db_session) -> int:
    """
    批量更新所有题目的难度（定时任务调用）
    返回更新数量
    """
    from ..models.models import QuestionBank

    questions = db_session.query(QuestionBank).filter(
        QuestionBank.status == 1
    ).all()

    updated = 0
    for q in questions:
        new_difficulty = get_difficulty_level(q.question_id, db_session)
        if q.difficulty != new_difficulty:
            q.difficulty = new_difficulty
            updated += 1

    if updated > 0:
        db_session.commit()
        logger.info(f"[difficulty] 更新 {updated} 题难度等级")

    return updated
```

def _scrape_exam_detail(url: str) -> dict:
    """
    进入考试详情页，抓取：
    - registration_start/end (报名时间)
    - exam_format (考试方式)
    - description (完整HTML内容)
    - images (二维码等图片)
    """
    result = {
        'registration_start': '',
        'registration_end': '',
        'exam_format': '',
        'description': '',
        'images': [],
    }
    try:
        resp = safe_request(url, timeout=15, label="exam/detail")
        if resp is None:
            return result
        resp.encoding = resp.apparent_encoding or 'utf-8'
        soup = BeautifulSoup(resp.text, 'lxml')

        # 提取正文
        for selector in [
            'div.article-content', 'div.content', 'div.article',
            'article', 'div.main', 'div.detail',
        ]:
            elem = soup.select_one(selector)
            if elem:
                # 移除脚本和样式
                for bad in elem.select('script, style, nav, footer'):
                    bad.decompose()
                result['description'] = str(elem)[:10000]
                break

        full_text = soup.get_text()

        # 提取报名时间
        import re
        time_patterns = [
            r'报名时间[：:]\s*(\d{4}[-/年]\d{1,2}[-/月]\d{1,2}[日]?\s*[-～~至]\s*\d{4}[-/年]?\d{1,2}[-/月]?\d{1,2}[日]?)',
            r'报名[：:]\s*(\d{1,2}月\d{1,2}日\s*[-～~至]\s*\d{1,2}月\d{1,2}日)',
        ]
        for pat in time_patterns:
            m = re.search(pat, full_text)
            if m:
                time_str = m.group(1)
                # 解析开始和结束时间
                parts = re.split(r'[-～~至]', time_str)
                if len(parts) == 2:
                    result['registration_start'] = parts[0].strip()
                    result['registration_end'] = parts[1].strip()
                break

        # 提取考试方式
        if '笔试' in full_text and '面试' in full_text:
            result['exam_format'] = '笔试+面试'
        elif '笔试' in full_text:
            result['exam_format'] = '笔试'
        elif '面试' in full_text:
            result['exam_format'] = '面试'

        # 提取图片（含二维码）
        for img in soup.select('img[src]'):
            src = img.get('src', '')
            if src and not src.startswith('data:'):
                if src.startswith('/'):
                    from urllib.parse import urljoin
                    src = urljoin(url, src)
                # 过滤小图标
                width = img.get('width', '0')
                if width and int(width) < 50:
                    continue
                result['images'].append(src)

        return result
    except Exception as e:
        logger.warning(f"[exam/detail] 详情抓取失败 {url}: {e}")
        return result
```

def _compute_status(e: ExamCalendarModel) -> str:
    """根据报名时间计算考试报名状态（精确判定）"""
    from datetime import datetime as dt
    now = dt.now()
    # 优先级：registration_end > registration_start > exam_date
    if e.registration_end and isinstance(e.registration_end, dt):
        if e.registration_end < now:
            return "已截止"
    if e.registration_start and isinstance(e.registration_start, dt):
        if e.registration_start > now:
            return "即将报名"
    if e.registration_start and e.registration_end \
       and isinstance(e.registration_start, dt) \
       and isinstance(e.registration_end, dt):
        if e.registration_start <= now <= e.registration_end:
            return "报名中"
    # 回退：根据exam_date判断
    if e.exam_date and e.exam_date < now:
        return "已截止"
    return "报名中"

def cleanup_expired_exams():
    """硬删除已截止且registration_end已过超过1天的考试记录"""
    from datetime import datetime as dt, timedelta
    db = SessionLocal()
    try:
        cutoff = dt.now() - timedelta(days=1)
        # 优先按registration_end删除，fallback用exam_date
        deleted = db.query(ExamCalendar).filter(
            (ExamCalendar.registration_end < cutoff) |
            ((ExamCalendar.registration_end == None) & (ExamCalendar.exam_date < cutoff))
        ).delete(synchronize_session=False)
        db.commit()
        if deleted > 0:
            logger.info(f"[scheduler] 清理过期考试: {deleted} 条")
        return deleted
    except Exception as e:
        db.rollback()
        logger.error(f"[scheduler] 清理过期失败: {e}")
        return 0
    finally:
        db.close()
```

def _extract_article_with_images(soup, page_url, news_id):
    """提取正文+图片，下载图片到磁盘"""
    import re, os, requests as req
    from urllib.parse import urljoin

    # 提取正文
    content = _extract_article_content(soup)

    # 提取图片
    images = []
    img_dir = f'uploads/news/{news_id}'
    os.makedirs(img_dir, exist_ok=True)

    for idx, img in enumerate(soup.select('img[src]')):
        src = img.get('src', '')
        if not src or src.startswith('data:'):
            continue
        # 跳过小图标
        w, h = img.get('width'), img.get('height')
        if w and h and (int(w) < 50 or int(h) < 50):
            continue
        full_url = urljoin(page_url, src)
        try:
            r = req.get(full_url, timeout=10, headers={'User-Agent': get_random_ua()})
            if r.status_code == 200:
                ext = full_url.rsplit('.', 1)[-1].split('?')[0] or 'jpg'

                    ext = 'jpg'
                fname = f'img_{idx}.{ext}'

                with open(fpath, 'wb') as f:

@router.post("/{news_id}/favorite", summary="收藏/取消收藏新闻")
async def toggle_favorite(
    news_id: int,
    current_user: SysUser = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """切换收藏状态：已收藏则取消，未收藏则添加"""
    existing = db.query(NewsFavorite).filter(
        NewsFavorite.user_id == current_user.user_id,
        NewsFavorite.news_id == news_id
    ).first()

    if existing:
        db.delete(existing)
        db.commit()
        return {"favorited": False, "message": "已取消收藏"}
    else:
        fav = NewsFavorite(user_id=current_user.user_id, news_id=news_id)
        db.add(fav)
        db.commit()
        return {"favorited": True, "message": "收藏成功"}


@router.get("/favorites", summary="获取收藏列表")
async def get_favorites(

methods: {
  async toggleFavorite(newsId) {
    const res = await window.__VTJ_AUTH__.apiFetch(`/api/news/${newsId}/favorite`, {

    this.state.showHighlightMenu = true;
    this.state.selectedText = text;

    const range = selection.getRangeAt(0);

      `/api/news/${this.state.selectedNews.id}/note`,

          note_text: '',
          quote_text: this.state.selectedText,
          quote_position: this.state.quotePosition,
          color: color
        })

    );

    result = {
        "content": "", "content_html": "", "organization": "",
        "requirements": "", "registration_start": "", "registration_end": "",
        "deadline": "", "region": "", "tag": "", "tag_color": "",

    try:
        # ... 现有逻辑 ...

        # 提取HTML正文
        for selector in content_selectors:
            elem = soup.select_one(selector)
            if elem:
                # 过滤：移除 script/style/nav/footer/.ad
                for bad in elem.select('script, style, nav, footer, .ad, .sidebar'):
                    bad.decompose()
                result["content_html"] = str(elem)[:10000]
                result["content"] = elem.get_text(separator="\n", strip=True)[:5000]

        # 提取图片
        for img in soup.select('img[src]'):
            src = img.get('src', '')
            if src and not src.startswith('data:'):
                full = urljoin(url, src) if not src.startswith('http') else src
                result["images"].append({"url": full, "alt": img.get('alt', '')})

        # 提取考试方式
        full_text = soup.get_text()
        if '笔试' in full_text and '面试' in full_text:
            result["exam_format"] = '笔试+面试'
        elif '笔试' in full_text:
            result["exam_format"] = '笔试'
        elif '面试' in full_text:
            result["exam_format"] = '面试'

        # 提取联系方式
        result["contact_info"] = _extract_field(full_text,
            ["联系电话", "联系方式", "咨询电话", "联系人"])

        return result
    except Exception as e:
