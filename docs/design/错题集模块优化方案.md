## 一、需求分析

### 需求1：错题自动录入（非硬编码）
- **用户预期**：用户在"快速练习"、"智能组卷"、"历年试卷"等做题模块中，系统比对完答案后自动将错题录入并分类到错题集
- **现状诊断**：
  - 错题列表已通过API获取（`GET /api/wrong-questions`），**非硬编码** ✓
  - 日常练习（answer_page.vue）已通过 `POST /api/wrong-questions/collect` 自动收录 ✓
  - **快速练习**（`POST /api/quick-practice/submit`）未写入错题表 ✗
  - **考试交卷**（`POST /api/exams/{record_id}/submit`）未写入错题表 ✗

### 需求2："斩"按钮 + 斩题10天过期 + 倒计时提醒
- **用户预期**：笔记左侧设计"斩"按钮，斩题保存到"我的斩题"，10天后硬删除，删除前2天提醒
- **现状诊断**：
  - 后端斩题接口已有（`PUT /api/wrong-questions/{wrong_id}/chop`），设置 `auto_delete_time=now+10天` ✓
  - 前端无"斩"按钮（只有"笔记"按钮）✗
  - 斩题列表 `zhantiList` 始终为空（第547行强制清空）✗
  - `confirmChopQuestion` 成功后未调用 `moveToZhanti` ✗
  - 无自动删除定时任务 ✗
  - 无倒计时提醒 ✗

### 需求3：日期格式化
- **用户预期**：显示为"年-月-日 时:分"格式
- **现状诊断**：第203行 `{{ q.date }}` 直接显示后端原始字符串（如 `2026-07-29T20:56:00.586180`）

def cleanup_expired_zhanti():
    """清理过期的斩题记录（auto_delete_time < now 的记录物理删除）"""
    from ..models import WrongQuestionPersonal
    from ..database import SessionLocal
    db = SessionLocal()
    try:
        expired = db.query(WrongQuestionPersonal).filter(
            WrongQuestionPersonal.is_cut == 1,
            WrongQuestionPersonal.auto_delete_time < datetime.now()
        ).all()
        for wq in expired:
            db.delete(wq)
        db.commit()
        if expired:
            logger.info(f"已清理 {len(expired)} 条过期斩题记录")
    except Exception as e:
        logger.error(f"清理斩题记录失败: {e}")
        db.rollback()
    finally:
        db.close()

scheduler.add_job(
    cleanup_expired_zhanti,
    trigger=CronTrigger(hour=3, minute=0, timezone=CST),
    id="cleanup_zhanti",
    name="每日清理过期斩题",
    replace_existing=True
)

# 接口1：获取斩题列表（含倒计时天数）
@router.get("/zhanti/list", summary="获取我的斩题列表")
async def get_zhanti_list(
    current_user: SysUser = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """获取已斩题列表，含剩余天数和过期提醒"""
    now = datetime.now()
    items = db.query(WrongQuestionPersonal).filter(
        WrongQuestionPersonal.user_id == current_user.user_id,
        WrongQuestionPersonal.is_cut == 1,
        WrongQuestionPersonal.status == 1
    ).all()

    result = []
    for wq in items:
        q = db.query(QuestionBank).filter(QuestionBank.question_id == wq.question_id).first()
        if not q:
            continue
        days_left = (wq.auto_delete_time - now).days if wq.auto_delete_time else 0
        result.append({
            'wrong_id': wq.wrong_id,
            'question_id': wq.question_id,
            'question_content': q.content[:100] if q.content else '',
            'cut_time': wq.cut_time.strftime('%Y-%m-%d %H:%M') if wq.cut_time else '',
            'auto_delete_time': wq.auto_delete_time.strftime('%Y-%m-%d %H:%M') if wq.auto_delete_time else '',
            'days_left': max(0, days_left),
            'is_expiring': days_left <= 2  # 2天内过期标记
        })
    return {'total': len(result), 'items': result}


# 接口2：获取即将过期的斩题提醒
@router.get("/zhanti/expiring", summary="获取即将过期的斩题提醒")
async def get_zhanti_expiring(
    current_user: SysUser = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """返回2天内即将过期的斩题数量和列表"""
    now = datetime.now()
    threshold = now + timedelta(days=2)
    items = db.query(WrongQuestionPersonal).filter(
        WrongQuestionPersonal.user_id == current_user.user_id,
        WrongQuestionPersonal.is_cut == 1,
        WrongQuestionPersonal.status == 1,
        WrongQuestionPersonal.auto_delete_time <= threshold,
        WrongQuestionPersonal.auto_delete_time > now
    ).all()

    return {
        'count': len(items),
        'message': '您有斩题进入观察期，请去看看' if items else '',
        'items': [{
            'wrong_id': wq.wrong_id,
            'days_left': (wq.auto_delete_time - now).days if wq.auto_delete_time else 0
        } for wq in items]
    }


# 接口3：取消斩题（恢复到错题列表）
@router.post("/{wrong_id}/unchop", summary="取消斩题，恢复到错题列表")
async def unchop_question(
    wrong_id: int,
    current_user: SysUser = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """取消斩题：is_cut=0, 清除auto_delete_time"""
    wq = db.query(WrongQuestionPersonal).filter(
        WrongQuestionPersonal.wrong_id == wrong_id,
        WrongQuestionPersonal.user_id == current_user.user_id
    ).first()
    if not wq:
        raise HTTPException(status_code=404, detail="错题记录不存在")

    wq.is_cut = 0
    wq.cut_time = None
    wq.auto_delete_time = None
    wq.mastery_level = "PARTIAL"
    db.commit()
    return {"message": "已恢复到错题列表", "wrong_id": wrong_id}
```

formatDate(dateStr) {
  if (!dateStr) return '';
  var d = new Date(dateStr);
  if (isNaN(d.getTime())) return dateStr;
  var year = d.getFullYear();
  var month = String(d.getMonth() + 1).padStart(2, '0');
  var day = String(d.getDate()).padStart(2, '0');
  var hours = String(d.getHours()).padStart(2, '0');
  var minutes = String(d.getMinutes()).padStart(2, '0');
  return year + '-' + month + '-' + day + ' ' + hours + ':' + minutes;
}

async chopQuestionDirect(q) {
  if (!q.wrong_id) {
    this.$message.warning('该题目无法斩题');
    return;
  }
  try {
    const res = await window.__VTJ_AUTH__.apiFetch('/api/wrong-questions/' + q.wrong_id + '/chop', {
      method: 'PUT',
      headers: this.getJsonHeaders()
    });
    if (res.ok) {
      q.chopped = true;
      q.status = 'chopped';
      q.statusText = '已斩题';
      this.$message.success('已斩题，10天后自动删除');
      this.loadZhantiList(); // 刷新斩题列表
    }
  } catch(e) {
    this.$message.error('斩题失败');
  }
},

async unchopQuestion(q) {
  if (!q.wrong_id) return;
  try {
    const res = await window.__VTJ_AUTH__.apiFetch('/api/wrong-questions/' + q.wrong_id + '/unchop', {
      method: 'POST',
      headers: this.getJsonHeaders()
    });
    if (res.ok) {
      q.chopped = false;
      q.status = 'needs-review';
      q.statusText = '待复习';
      this.$message.success('已恢复到错题列表');
      this.loadZhantiList();
    }
  } catch(e) {
    this.$message.error('恢复失败');
  }
}
```

async loadZhantiList() {
  try {
    const res = await window.__VTJ_AUTH__.apiFetch('/api/wrong-questions/zhanti/list', {
      headers: this.getJsonHeaders()
    });
    const data = await res.json();
    this.state.zhantiList = (data.items || []).map(function(item) {
      return {
        id: item.wrong_id,
        title: item.question_content,
        createTime: item.cut_time,
        daysLeft: item.days_left,
        autoDeleteTime: item.auto_delete_time,
        isExpiring: item.is_expiring,
        originalQuestionId: item.question_id
      };
    });
  } catch(e) {
    console.warn('加载斩题列表失败:', e);
    this.state.zhantiList = [];
  }
},

async loadZhantiExpiring() {
  try {
    const res = await window.__VTJ_AUTH__.apiFetch('/api/wrong-questions/zhanti/expiring', {
      headers: this.getJsonHeaders()
    });
    const data = await res.json();
    this.state.zhantiExpiringNotice = data.count > 0;
    this.state.zhantiExpiringMessage = data.message;
    this.state.zhantiExpiringCount = data.count;
  } catch(e) {
    this.state.zhantiExpiringNotice = false;
  }
}

## 四、数据流设计

### 4.1 错题自动录入流程

```
快速练习                          智能组卷/历年试卷（考试）
  │                                    │
  ▼                                    ▼
POST /api/quick-practice/submit    POST /api/exams/{id}/submit
  │                                    │
  ├─ 写入 PracticeRecord              ├─ 自动评分
  └─ 答错 → _auto_record_wrong()      └─ 答错 → _auto_record_wrong()
        └─ 写入 WrongQuestionPersonal       └─ 写入 WrongQuestionPersonal
              (collection_type="WRONG")           (去重: 已存在则error_count+1)
```

### 4.2 斩题完整生命周期

```
错题列表 → 点击"斩"按钮
  │
  ▼
PUT /api/wrong-questions/{id}/chop
  ├─ is_cut=1
  ├─ cut_time=now
  └─ auto_delete_time=now+10天
  │
  ▼
"我的斩题"列表显示（含剩余天数）
  │
  ├─ 用户点击"恢复" → POST /api/wrong-questions/{id}/unchop
  │    └─ is_cut=0, auto_delete_time=null → 回到错题列表
  │
  └─ 10天后 → 定时任务(每日03:00)物理删除
       │
       └─ 删除前2天 → GET /api/wrong-questions/zhanti/expiring
            └─ 前端显示提醒条："您有斩题进入观察期，请去看看"
```
