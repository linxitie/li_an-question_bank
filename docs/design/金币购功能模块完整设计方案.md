- **文档版本**：v2.0
- **编写日期**：2026年9月
- **优化说明**：在v1.0基础上完成手机端、平板端、电脑端三端适配设计，深化页面布局与交互细节，保留所有可落地内容并完善实际功能体验。
- **适用对象**：前端开发人员、UI/UX设计师、后端开发人员


## 一、功能定位与整体架构

### 1.1 功能定位

“金币购”是砺安题库系统**学习激励与用户留存**的核心模块，通过虚拟金币与实物/虚拟奖励的结合，构建“学习→赚取金币→兑换奖励”的正向循环。

### 1.2 三端功能对比

| 功能模块 | 电脑端 | 平板端 | 手机端 |
|---------|--------|--------|--------|
| 金币商城首页 | ✅ 双栏布局 | ✅ 顶部Tab+卡片流 | ✅ 单列卡片流 |
| 商品详情 | ✅ 弹窗/独立页 | ✅ 弹窗 | ✅ 底部抽屉 |
| 充值中心 | ✅ 弹窗 | ✅ 弹窗 | ✅ 底部抽屉 |
| 金币明细 | ✅ 独立页 | ✅ 独立页 | ✅ 独立页 |
| 任务中心 | ✅ 独立页 | ✅ 独立页 | ✅ 独立页 |
| 个人目标 | ✅ 独立页 | ✅ 独立页 | ✅ 独立页 |
| 成就体系 | ✅ 独立页 | ✅ 独立页 | ✅ 独立页 |

```python
# backend/app/services/coin_service.py

RECHARGE_TIERS = [
    {"rmb": 6, "coins": 60, "bonus_rate": 0.00},
    {"rmb": 30, "coins": 330, "bonus_rate": 0.10},
    {"rmb": 68, "coins": 816, "bonus_rate": 0.20},
    {"rmb": 128, "coins": 1664, "bonus_rate": 0.30},
    {"rmb": 328, "coins": 4592, "bonus_rate": 0.40},
    {"rmb": 648, "coins": 9720, "bonus_rate": 0.50},
]

def calculate_recharge_coins(rmb_amount: int) -> dict:
    """计算充值所得砺安币"""
    tier = next((t for t in RECHARGE_TIERS if t["rmb"] == rmb_amount), None)
    if not tier:
        raise ValueError("不支持的充值金额")
    
    base_coins = tier["coins"]
    bonus_coins = int(base_coins * tier["bonus_rate"])
    total_coins = base_coins + bonus_coins
    
    return {
        "base_coins": base_coins,
        "bonus_coins": bonus_coins,
        "total_coins": total_coins,
        "tier_name": f"{rmb_amount}元档"
    }
```

### 5.2 个人目标完成度精准判定算法

```python
# backend/app/services/goal_service.py

class GoalCompletionEvaluator:
    """
    个人目标完成度精准判定算法
    基于多维度加权计算，精准评估用户目标完成情况
    """
    
    def evaluate_goal_completion(self, user_id: int, goal: Dict) -> Dict:
        """
        评估用户个人目标的完成度
        返回: { "completion_rate": float, "quality_score": float, "reward_coins": int }
        """
        # 维度1：完成率（权重40%）
        completion_rate = self._calc_completion_rate(user_id, goal)
        
        # 维度2：完成质量（权重25%）
        quality_score = self._calc_quality_score(user_id, goal)
        
        # 维度3：坚持天数（权重20%）
        persistence_score = self._calc_persistence_score(user_id, goal)
        
        # 维度4：进步幅度（权重15%）
        improvement_score = self._calc_improvement_score(user_id, goal)
        
        # 综合完成度评分（0-1之间）
        overall_completion = (
            completion_rate * 0.40 +
            quality_score * 0.25 +
            persistence_score * 0.20 +
            improvement_score * 0.15
        )
        
        # 目标难度系数
        difficulty_factor = self._calc_difficulty_factor(goal)
        
        # 基础奖励 × 完成度 × 难度系数
        base_reward = goal.get("base_reward", 50)
        reward_coins = int(base_reward * overall_completion * difficulty_factor)
        
        return {
            "completion_rate": completion_rate,
            "quality_score": quality_score,
            "persistence_score": persistence_score,
            "improvement_score": improvement_score,
            "overall_completion": overall_completion,
            "difficulty_factor": difficulty_factor,
            "reward_coins": max(reward_coins, 10)  # 最低保底10金币
        }
    
    def _calc_completion_rate(self, user_id: int, goal: Dict) -> float:
        """计算完成率"""
        target = goal.get("target_value", 0)
        actual = self._get_user_actual_progress(user_id, goal)
        if target == 0:
            return 0.0
        return min(actual / target, 1.0)
    
    def _calc_quality_score(self, user_id: int, goal: Dict) -> float:
        """计算完成质量（正确率、效率等）"""
        accuracy = self._get_user_accuracy_in_scope(user_id, goal)
        if accuracy >= 0.85:
            return 1.0
        elif accuracy >= 0.70:
            return 0.7
        elif accuracy >= 0.60:
            return 0.5
        else:
            return 0.3
    
    def _calc_persistence_score(self, user_id: int, goal: Dict) -> float:
        """计算坚持天数得分"""
        consecutive_days = self._get_consecutive_days(user_id, goal)
        if consecutive_days >= 30:
            return 1.0
        elif consecutive_days >= 21:
            return 0.9
        elif consecutive_days >= 14:
            return 0.75
        elif consecutive_days >= 7:
            return 0.6
        elif consecutive_days >= 3:
            return 0.4
        else:
            return 0.2
    
    def _calc_improvement_score(self, user_id: int, goal: Dict) -> float:
        """计算进步幅度"""
        before = self._get_user_performance_before(user_id, goal)
        after = self._get_user_performance_during(user_id, goal)
        improvement = (after - before) / max(before, 1)
        return min(max(improvement, 0), 1.0)
    
    def _calc_difficulty_factor(self, goal: Dict) -> float:
        """计算目标难度系数"""
        target_value = goal.get("target_value", 0)
        goal_type = goal.get("goal_type", "")
        
        base_factors = {
            "daily_questions": 1.0 + (target_value / 100) * 0.5,
            "weekly_questions": 1.0 + (target_value / 500) * 0.5,
            "accuracy_goal": 1.0 + (target_value - 0.6) * 2.5,
            "continuous_learning": 1.0 + (target_value / 30) * 0.5,
            "special_breakthrough": 1.2 + (target_value / 100) * 0.3,
        }
        factor = base_factors.get(goal_type, 1.0)
        return min(factor, 3.0)
```

```python
# backend/app/services/achievement_service.py

class AchievementEngine:
    """
    成就规则引擎
    负责：事件监听 → 规则匹配 → 成就解锁 → 奖励发放 → 幂等保证
    """
    
    ACHIEVEMENT_RULES = {
        "ACH_LEARN_001": {
            "name": "初出茅庐",
            "category": "learn",
            "condition": lambda ctx: ctx["total_questions"] >= 1,
            "reward": 10,
            "icon": "🌱"
        },
        "ACH_PERSIST_002": {
            "name": "连续学习7天",
            "category": "persist",
            "condition": lambda ctx: ctx["continuous_days"] >= 7,
            "reward": 40,
            "icon": "🥇"
        },
        # ... 其他46枚成就
    }
    
    async def process_event(self, user_id: int, event_type: str, event_data: Dict):
        """处理用户行为事件，触发成就检查"""
        context = await self._build_user_context(user_id)
        
        for achievement_id, rule in self.ACHIEVEMENT_RULES.items():
            try:
                if rule["condition"](context):
                    unlocked = await self._unlock_achievement(
                        user_id=user_id,
                        achievement_id=achievement_id,
                        rule=rule
                    )
                    if unlocked:
                        logger.info(f"[成就解锁] user={user_id} achievement={rule['name']}")
            except Exception as e:
                logger.error(f"成就检查失败: {achievement_id}, error={e}")
                continue
    
    async def _unlock_achievement(self, user_id: int, achievement_id: str, rule: Dict) -> bool:
        """解锁成就（幂等操作）"""
        async with db.transaction():
            try:
                # 利用唯一索引保证幂等
                stmt = insert(UserAchievement).values(
                    user_id=user_id,
                    achievement_id=achievement_id,
                    achievement_name=rule["name"],
                    category=rule["category"],
                    icon=rule["icon"],
                    reward_coins=rule["reward"],
                    obtained_at=datetime.now()
                )
                await db.execute(stmt)
                
                # 发放金币
                await self._grant_coins(
                    user_id=user_id,
                    amount=rule["reward"],
                    source="achievement",
                    source_id=achievement_id,
                    description=f"解锁成就：{rule['name']}"
                )
                
                # 发送通知
                await self._send_notification(
                    user_id=user_id,
                    title=f"🏆 解锁新成就：{rule['name']}",
                    content=f"恭喜你获得「{rule['name']}」成就，奖励 {rule['reward']} 金币！",
                    type="achievement"
                )
                
                await db.commit()
                return True
                
            except IntegrityError:
                await db.rollback()
                return False
            except Exception as e:
                await db.rollback()
                logger.error(f"成就解锁失败: {e}")
                raise
```
