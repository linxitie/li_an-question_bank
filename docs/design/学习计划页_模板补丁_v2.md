### 1.2 script 增补（`<script setup>` 内）

```js
      // state 里加：schedule: null, adjustNotice: ''
      // loadToday() 里 state.today = await res.json() 之后补一行：
      //   state.schedule = state.today.schedule || null

      async function toggleScheduleItem(item) {
        if (!item || !item.goal_id) return
        try {
          const res = await window.__VTJ_AUTH__.apiFetch(`/api/plan/task/${item.goal_id}/toggle`, { method: 'POST' })
          if (!res.ok) return
          await loadToday()
          const cur = await window.__VTJ_AUTH__.apiFetch('/api/plan/current')
          if (cur.ok) state.plan = await cur.json()
        } catch (e) { /* 单次失败不影响其它区块 */ }
      }

      const schedulePercent = computed(() => {
        const s = state.schedule
        if (!s || !s.slots) return 0
        let total = 0, done = 0
        s.slots.forEach(p => p.items.forEach(it => {
          total += it.minutes || 0
          if (it.done) done += it.minutes || 0
        }))
        return total ? Math.round(done / total * 100) : 0
      })
```

script：

```js
      // askAI 成功后：state.adjustResult = data; 并把 state.adjustHistory = state.plan.recent_adjustments
      async function undoAdjust() {
        try {
          const res = await window.__VTJ_AUTH__.apiFetch('/api/plan/adjust', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ instruction: '', undo: true })
          })
          const data = await res.json().catch(() => ({}))
          if (res.ok) {
            state.adjustNotice = data.message || '已撤销'
            const cur = await window.__VTJ_AUTH__.apiFetch('/api/plan/current')
            if (cur.ok) state.plan = await cur.json()
            await loadToday()
          }
        } catch (e) { /* 撤销失败保持原状 */ }
      }
```

### 2.1 面试面板加作答区（替换 `:292` 附近的题目展示区之后）

```html
        <div class="interview-answer">

            <span class="interview-timer">已用时 {{ state.interviewSeconds }}s</span>

```

script（多轮 + 评分 + 语音）：

```js
      // state 增：interviewAnswer: '', interviewRounds: 0, interviewSeconds: 0,
      //           interviewGrading: false, interviewStartAt: 0, interviewScores: null
      function startInterviewTimer() {
        if (state.interviewTimer) clearInterval(state.interviewTimer)
        state.interviewStartAt = Date.now()
        state.interviewTimer = setInterval(() => {
          state.interviewSeconds = Math.floor((Date.now() - state.interviewStartAt) / 1000)
        }, 1000)
      }

      async function submitInterviewAnswer() {
        if (!state.interviewAnswer.trim()) return
        state.interviewGrading = true
        try {
          const round = state.interviewRounds + 1
          const isLast = round >= 3          // 出题 + 追问 2 轮
          const prompt = isLast
            ? `【评分】面试题：${state.interviewQuestion}\n考生作答：${state.interviewAnswer}\n`
              + '请只输出 JSON：{"scores":{"content":0-20,"logic":0-20,"expression":0-20,"manner":0-20},'
              + '"total":0-80,"advice":["改进点1","改进点2"]}'
            : `【追问】面试题：${state.interviewQuestion}\n考生作答：${state.interviewAnswer}\n`
              + '请以考官口吻追问一个问题（一句话，不超过 40 字），只输出追问文本。'
          const reply = await callAiOneShot(prompt, 'interview')
          state.interviewRounds = round
          if (isLast) {
            try { state.interviewScores = JSON.parse(reply) } catch (e) { state.interviewScores = { raw: reply } }
          } else {
            state.interviewQuestion = String(reply).trim()
          }
          state.interviewAnswer = ''
        } finally { state.interviewGrading = false }
      }

      function startVoiceAnswer() {
        const SR = window.SpeechRecognition || window.webkitSpeechRecognition
        if (!SR) { state.error = '当前浏览器不支持语音识别，请手动输入'; return }
        const rec = new SR()
        rec.lang = 'zh-CN'; rec.interimResults = false; rec.maxAlternatives = 1
        state.error = '正在听……'
        rec.onresult = (e) => {
          state.interviewAnswer = (state.interviewAnswer + ' ' + e.results[0][0].transcript).trim()
          state.error = ''
        }
        rec.onerror = () => { state.error = '语音识别失败，请改用文字' }
        rec.start()
      }
```

> 语音用浏览器原生 `SpeechRecognition`（Chrome/Edge 支持）；不支持时按钮提示改文字。
> 若要做"考官朗读题目"，可加：`new SpeechSynthesisUtterance(state.interviewQuestion)` + `speechSynthesis.speak()`。
