## 3. `api_surface` 端点基线

python -c "
import json,collections
d=json.load(open(r'backend/data/refactor_baseline/api_surface.json',encoding='utf-8'))
tot=0; bydom=collections.Counter()
for f,v in d.items():
    c=v.get('endpoint_count',0); tot+=c; bydom[v.get('domain')]+=c
print('sum endpoint_count =',tot)
print('actual endpoints list len =',sum(len(v.get('endpoints',[])) for v in d.values()))
print('modules =',len(d))
for k,v in sorted(bydom.items(), key=lambda x:-x[1]): print(f'{k}\t{v}')
